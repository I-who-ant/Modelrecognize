```python
import numpy as np
import matplotlib.pyplot as plt

def estimate_required_data_size(task_complexity, target_accuracy):
    """
    估算所需数据量

    Args:
        task_complexity: 'simple', 'medium', 'complex'
        target_accuracy: 目标准确率 (0-1)

    Returns:
        estimated_size: 估算的数据量
    """

    # 基础数据量
    base_sizes = {
        'simple': 1000,
        'medium': 10000,
        'complex': 100000,
    }

    base = base_sizes.get(task_complexity, 10000)

    # 准确率调整系数
    # 准确率越高,需要的数据越多
    if target_accuracy < 0.8:
        multiplier = 1.0
    elif target_accuracy < 0.9:
        multiplier = 2.0
    elif target_accuracy < 0.95:
        multiplier = 5.0
    else:
        multiplier = 10.0

    estimated_size = int(base * multiplier)

    print(f"任务复杂度: {task_complexity}")
    print(f"目标准确率: {target_accuracy:.0%}")
    print(f"估算所需数据量: {estimated_size:,} 样本")

    return estimated_size
```

### 第4步: 数据预处理 🧹

#### 4.1 处理缺失值

```python
"""
缺失值处理策略:

1. 删除:
   - 缺失率 > 50%: 删除该特征
   - 样本缺失过多: 删除该样本

2. 填充:
   - 数值型:
     * 均值/中位数 (适合正态分布)
     * 众数 (适合偏态分布)
     * 前向/后向填充 (时序数据)
     * 插值 (时序数据)

   - 类别型:
     * 众数
     * 新类别 'missing'
     * 根据其他特征预测

3. 标记:
   - 增加一个二值特征标记是否缺失
"""

class DataPreprocessor:
    """数据预处理器"""

    def __init__(self):
        self.imputers = {}
        self.scalers = {}
        self.encoders = {}

    def handle_missing_values(self, df, strategy='auto'):
        """
        处理缺失值

        Args:
            df: 数据框
            strategy: 'auto', 'drop', 'mean', 'median', 'mode'
        """
        df_clean = df.copy()

        # 自动策略
        if strategy == 'auto':
            # 删除缺失率>50%的特征
            missing_percent = df.isnull().sum() / len(df)
            cols_to_drop = missing_percent[missing_percent > 0.5].index
            if len(cols_to_drop) > 0:
                print(f"删除缺失率>50%的特征: {list(cols_to_drop)}")
                df_clean = df_clean.drop(columns=cols_to_drop)

            # 数值型: 用中位数填充
            numeric_cols = df_clean.select_dtypes(include=[np.number]).columns
            for col in numeric_cols:
                if df_clean[col].isnull().any():
                    median_value = df_clean[col].median()
                    df_clean[col].fillna(median_value, inplace=True)
                    self.imputers[col] = median_value
                    print(f"{col}: 用中位数 {median_value:.2f} 填充")

            # 类别型: 用众数填充
            categorical_cols = df_clean.select_dtypes(include=['object']).columns
            for col in categorical_cols:
                if df_clean[col].isnull().any():
                    mode_value = df_clean[col].mode()[0]
                    df_clean[col].fillna(mode_value, inplace=True)
                    self.imputers[col] = mode_value
                    print(f"{col}: 用众数 '{mode_value}' 填充")

        return df_clean

    def detect_outliers(self, df, method='iqr'):
        """
        检测异常值

        Args:
            method: 'iqr' (四分位数) 或 'zscore' (标准分数)
        """
        outliers = {}
        numeric_cols = df.select_dtypes(include=[np.number]).columns

        for col in numeric_cols:
            if method == 'iqr':
                Q1 = df[col].quantile(0.25)
                Q3 = df[col].quantile(0.75)
                IQR = Q3 - Q1
                lower_bound = Q1 - 1.5 * IQR
                upper_bound = Q3 + 1.5 * IQR

                outlier_mask = (df[col] < lower_bound) | (df[col] > upper_bound)
                outliers[col] = df[outlier_mask]

                if outlier_mask.sum() > 0:
                    print(f"{col}: 发现 {outlier_mask.sum()} 个异常值 "
                          f"(范围: [{lower_bound:.2f}, {upper_bound:.2f}])")

            elif method == 'zscore':
                z_scores = np.abs((df[col] - df[col].mean()) / df[col].std())
                outlier_mask = z_scores > 3
                outliers[col] = df[outlier_mask]

                if outlier_mask.sum() > 0:
                    print(f"{col}: 发现 {outlier_mask.sum()} 个异常值 "
                          f"(|z-score| > 3)")

        return outliers

    def scale_features(self, df, method='standard'):
        """
        特征缩放

        Args:
            method: 'standard' (标准化) 或 'minmax' (归一化)
        """
        from sklearn.preprocessing import StandardScaler, MinMaxScaler

        df_scaled = df.copy()
        numeric_cols = df.select_dtypes(include=[np.number]).columns

        if method == 'standard':
            scaler = StandardScaler()
        else:
            scaler = MinMaxScaler()

        df_scaled[numeric_cols] = scaler.fit_transform(df[numeric_cols])
        self.scalers['numeric'] = scaler

        print(f"特征缩放完成 (方法: {method})")
        return df_scaled

    def encode_categorical(self, df, method='onehot'):
        """
        类别特征编码

        Args:
            method: 'onehot', 'label', 'target'
        """
        from sklearn.preprocessing import OneHotEncoder, LabelEncoder

        df_encoded = df.copy()
        categorical_cols = df.select_dtypes(include=['object']).columns

        if method == 'onehot':
            # One-Hot编码
            df_encoded = pd.get_dummies(df_encoded,
                                       columns=categorical_cols,
                                       prefix=categorical_cols)
            print(f"One-Hot编码完成,新增 {len(df_encoded.columns) - len(df.columns)} 个特征")

        elif method == 'label':
            # Label编码
            for col in categorical_cols:
                le = LabelEncoder()
                df_encoded[col] = le.fit_transform(df[col])
                self.encoders[col] = le
            print(f"Label编码完成")

        return df_encoded
```

### 第5步: 特征工程 🛠️

```python
"""
特征工程: 从原始数据中提取有用的特征

类型:
1. 特征提取: 从原始数据提取新特征
2. 特征选择: 选择最有用的特征
3. 特征构造: 组合现有特征创建新特征
4. 特征变换: 改变特征的分布
"""

class FeatureEngineer:
    """特征工程工具"""

    def create_polynomial_features(self, df, degree=2):
        """
        创建多项式特征

        例: [x1, x2] → [x1, x2, x1², x1*x2, x2²]
        """
        from sklearn.preprocessing import PolynomialFeatures

        numeric_cols = df.select_dtypes(include=[np.number]).columns

        poly = PolynomialFeatures(degree=degree, include_bias=False)
        poly_features = poly.fit_transform(df[numeric_cols])

        # 创建特征名
        feature_names = poly.get_feature_names_out(numeric_cols)

        df_poly = pd.DataFrame(poly_features,
                              columns=feature_names,
                              index=df.index)

        print(f"创建多项式特征 (degree={degree})")
        print(f"原始特征数: {len(numeric_cols)}")
        print(f"新特征数: {len(feature_names)}")

        return df_poly

    def create_interaction_features(self, df, pairs):
        """
        创建交互特征

        Args:
            pairs: 特征对列表 [('f1', 'f2'), ('f3', 'f4')]
        """
        df_new = df.copy()

        for f1, f2 in pairs:
            # 乘积
            df_new[f'{f1}_x_{f2}'] = df[f1] * df[f2]

            # 比值 (避免除零)
            df_new[f'{f1}_div_{f2}'] = df[f1] / (df[f2] + 1e-8)

            # 差值
            df_new[f'{f1}_sub_{f2}'] = df[f1] - df[f2]

        print(f"创建 {len(pairs)} 对交互特征")
        return df_new

    def select_features_by_correlation(self, df, target_col, threshold=0.1):
        """
        根据相关性选择特征
        """
        numeric_cols = df.select_dtypes(include=[np.number]).columns
        numeric_cols = [c for c in numeric_cols if c != target_col]

        # 计算与目标的相关性
        correlations = df[numeric_cols].corrwith(df[target_col]).abs()

        # 选择相关性高的特征
        selected = correlations[correlations > threshold].sort_values(ascending=False)

        print(f"相关性阈值: {threshold}")
        print(f"选择的特征数: {len(selected)} / {len(numeric_cols)}")
        print("\n前10个特征:")
        print(selected.head(10))

        return selected.index.tolist()

    def select_features_by_importance(self, df, target_col, top_k=10):
        """
        根据特征重要性选择特征
        """
        from sklearn.ensemble import RandomForestClassifier

        X = df.drop(columns=[target_col])
        y = df[target_col]

        # 训练随机森林获取特征重要性
        rf = RandomForestClassifier(n_estimators=100, random_state=42)
        rf.fit(X, y)

        # 特征重要性
        importances = pd.Series(rf.feature_importances_,
                               index=X.columns).sort_values(ascending=False)

        selected = importances.head(top_k)

        print(f"选择Top-{top_k}重要特征:")
        print(selected)

        # 可视化
        plt.figure(figsize=(10, 6))
        selected.plot(kind='barh')
        plt.xlabel('重要性')
        plt.title(f'Top-{top_k} 特征重要性')
        plt.gca().invert_yaxis()
        plt.tight_layout()
        plt.show()

        return selected.index.tolist()
```

### 第6步: 模型选择与训练 🤖

```python
"""
模型选择指南:

任务类型 → 推荐模型

分类任务:
├─→ 线性可分 → Logistic Regression, Linear SVM
├─→ 非线性 → Random Forest, XGBoost, Neural Network
├─→ 小数据 → SVM, Decision Tree
├─→ 大数据 → Neural Network, XGBoost
└─→ 高维稀疏 → Logistic Regression, Linear SVM

回归任务:
├─→ 线性关系 → Linear Regression, Ridge, Lasso
├─→ 非线性 → Random Forest, XGBoost, Neural Network
└─→ 时序数据 → LSTM, GRU, Transformer

从简单到复杂:
1. 基线模型: Logistic Regression / Linear Regression
2. 集成模型: Random Forest / XGBoost
3. 深度学习: Neural Network / Transformer
"""

class ModelTrainer:
    """模型训练器"""

    def train_baseline_model(self, X_train, y_train, X_val, y_val, task='classification'):
        """
        训练基线模型
        """
        from sklearn.linear_model import LogisticRegression, LinearRegression
        from sklearn.metrics import accuracy_score, mean_squared_error

        print("=" * 60)
        print("训练基线模型")
        print("=" * 60)

        if task == 'classification':
            model = LogisticRegression(max_iter=1000)
            model.fit(X_train, y_train)

            train_pred = model.predict(X_train)
            val_pred = model.predict(X_val)

            train_acc = accuracy_score(y_train, train_pred)
            val_acc = accuracy_score(y_val, val_pred)

            print(f"训练集准确率: {train_acc:.4f}")
            print(f"验证集准确率: {val_acc:.4f}")
            print(f"Gap: {train_acc - val_acc:.4f}")

            # 诊断
            if train_acc < 0.8:
                print("⚠️  高偏差(欠拟合) - 建议: 增加模型容量")
            elif train_acc - val_acc > 0.1:
                print("⚠️  高方差(过拟合) - 建议: 增加正则化")
            else:
                print("✅ 模型表现良好")

        else:  # regression
            model = LinearRegression()
            model.fit(X_train, y_train)

            train_pred = model.predict(X_train)
            val_pred = model.predict(X_val)

            train_mse = mean_squared_error(y_train, train_pred)
            val_mse = mean_squared_error(y_val, val_pred)

            print(f"训练集MSE: {train_mse:.4f}")
            print(f"验证集MSE: {val_mse:.4f}")

        return model

    def train_tree_model(self, X_train, y_train, X_val, y_val):
        """
        训练树模型 (Random Forest)
        """
        from sklearn.ensemble import RandomForestClassifier
        from sklearn.metrics import accuracy_score, classification_report

        print("=" * 60)
        print("训练随机森林")
        print("=" * 60)

        model = RandomForestClassifier(
            n_estimators=100,
            max_depth=10,
            min_samples_split=10,
            random_state=42
        )

        model.fit(X_train, y_train)

        train_pred = model.predict(X_train)
        val_pred = model.predict(X_val)

        train_acc = accuracy_score(y_train, train_pred)
        val_acc = accuracy_score(y_val, val_pred)

        print(f"训练集准确率: {train_acc:.4f}")
        print(f"验证集准确率: {val_acc:.4f}")
        print(f"Gap: {train_acc - val_acc:.4f}")
        print()

        print("验证集详细报告:")
        print(classification_report(y_val, val_pred))

        return model

    def train_xgboost_model(self, X_train, y_train, X_val, y_val):
        """
        训练XGBoost
        """
        import xgboost as xgb
        from sklearn.metrics import accuracy_score

        print("=" * 60)
        print("训练XGBoost")
        print("=" * 60)

        # 创建DMatrix
        dtrain = xgb.DMatrix(X_train, label=y_train)
        dval = xgb.DMatrix(X_val, label=y_val)

        # 参数
        params = {
            'objective': 'binary:logistic',
            'eval_metric': 'logloss',
            'max_depth': 6,
            'learning_rate': 0.1,
            'subsample': 0.8,
            'colsample_bytree': 0.8,
        }

        # 训练
        evals = [(dtrain, 'train'), (dval, 'val')]
        model = xgb.train(
            params,
            dtrain,
            num_boost_round=100,
            evals=evals,
            early_stopping_rounds=10,
            verbose_eval=10

        # 评估
        train_pred = (model.predict(dtrain) > 0.5).astype(int)
        val_pred = (model.predict(dval) > 0.5).astype(int)

        val_acc = accuracy_score(y_val, val_pred)

        print(f"\n最终结果:")
        print(f"训练集准确率: {train_acc:.4f}")
        print(f"验证集准确率: {val_acc:.4f}")

        return model
```

### 第7步: 模型评估 📈

```python
"""
评估指标选择:

分类任务:
├─→ 准确率 (Accuracy): 整体正确率
├─→ 精确率 (Precision): 预测为正的有多少是对的
├─→ 召回率 (Recall): 真正为正的有多少被找出
├─→ F1分数: 精确率和召回率的调和平均
├─→ AUC-ROC: 综合评估分类能力
└─→ 混淆矩阵: 详细的分类结果

回归任务:
├─→ MSE (均方误差): 平均平方误差
├─→ RMSE (均方根误差): MSE的平方根
├─→ MAE (平均绝对误差): 平均绝对差距
└─→ R² (决定系数): 解释的方差比例
"""

class ModelEvaluator:
    """模型评估器"""

    def evaluate_classification(self, y_true, y_pred, y_prob=None):
        """
        分类任务评估
        """
        from sklearn.metrics import (accuracy_score, precision_score,
                                     recall_score, f1_score,
                                     confusion_matrix, classification_report,
                                     roc_auc_score, roc_curve)

        print("=" * 60)
        print("分类模型评估")
        print("=" * 60)
        print()

        # 1. 基本指标
        accuracy = accuracy_score(y_true, y_pred)
        precision = precision_score(y_true, y_pred, average='weighted')
        recall = recall_score(y_true, y_pred, average='weighted')
        f1 = f1_score(y_true, y_pred, average='weighted')

        print("基本指标:")
        print(f"  准确率 (Accuracy):  {accuracy:.4f}")
        print(f"  精确率 (Precision): {precision:.4f}")
        print(f"  召回率 (Recall):    {recall:.4f}")
        print(f"  F1分数:            {f1:.4f}")
        print()

        # 2. 混淆矩阵
        cm = confusion_matrix(y_true, y_pred)
        print("混淆矩阵:")
        print(cm)
        print()

        # 可视化混淆矩阵
        plt.figure(figsize=(8, 6))
        sns.heatmap(cm, annot=True, fmt='d', cmap='Blues')
        plt.title('混淆矩阵')
        plt.ylabel('真实标签')
        plt.xlabel('预测标签')
        plt.tight_layout()
        plt.show()

        # 3. 详细报告
        print("详细报告:")
        print(classification_report(y_true, y_pred))

        # 4. ROC-AUC (如果有概率)
        if y_prob is not None:
            auc = roc_auc_score(y_true, y_prob)
            print(f"AUC-ROC: {auc:.4f}")

            # 绘制ROC曲线
            fpr, tpr, thresholds = roc_curve(y_true, y_prob)

            plt.figure(figsize=(8, 6))
            plt.plot(fpr, tpr, linewidth=2, label=f'ROC (AUC={auc:.3f})')
            plt.plot([0, 1], [0, 1], 'k--', label='随机猜测')
            plt.xlabel('假正率 (FPR)')
            plt.ylabel('真正率 (TPR)')
            plt.title('ROC曲线')
            plt.legend()
            plt.grid(True, alpha=0.3)
            plt.tight_layout()
            plt.show()

        return {
            'accuracy': accuracy,
            'precision': precision,
            'recall': recall,
            'f1': f1,
        }

    def evaluate_regression(self, y_true, y_pred):
        """
        回归任务评估
        """
        from sklearn.metrics import mean_squared_error, mean_absolute_error, r2_score

        print("=" * 60)
        print("回归模型评估")
        print("=" * 60)
        print()

        # 计算指标
        mse = mean_squared_error(y_true, y_pred)
        rmse = np.sqrt(mse)
        mae = mean_absolute_error(y_true, y_pred)
        r2 = r2_score(y_true, y_pred)

        print("评估指标:")
        print(f"  MSE (均方误差):      {mse:.4f}")
        print(f"  RMSE (均方根误差):   {rmse:.4f}")
        print(f"  MAE (平均绝对误差):  {mae:.4f}")
        print(f"  R² (决定系数):       {r2:.4f}")
        print()

        # 可视化
        fig, axes = plt.subplots(1, 2, figsize=(14, 5))

        # 1. 预测vs真实
        axes[0].scatter(y_true, y_pred, alpha=0.5)
        axes[0].plot([y_true.min(), y_true.max()],
                    [y_true.min(), y_true.max()],
                    'r--', linewidth=2, label='完美预测')
        axes[0].set_xlabel('真实值')
        axes[0].set_ylabel('预测值')

        axes[0].legend()
        axes[0].grid(True, alpha=0.3)

        # 2. 残差分布
        residuals = y_true - y_pred
        axes[1].hist(residuals, bins=50, edgecolor='black')
        axes[1].axvline(x=0, color='r', linestyle='--', linewidth=2)
        axes[1].set_xlabel('残差 (真实 - 预测)')
        axes[1].set_ylabel('频数')
        axes[1].set_title('残差分布')
        axes[1].grid(True, alpha=0.3)

        plt.tight_layout()
        plt.show()

        return {
            'mse': mse,
            'rmse': rmse,
            'mae': mae,
            'r2': r2,
        }
```

### 第11步: 部署与监控 🚀

```python
"""
模型部署流程:

1. 模型序列化
   - 保存模型
   - 保存预处理器
   - 记录版本信息

2. 部署方式
   - 批量预测
   - 在线API
   - 边缘部署

3. 监控指标
   - 预测性能
   - 响应时间
   - 资源占用
   - 数据漂移
"""

class ModelDeployer:
    """模型部署器"""

    def save_model(self, model, preprocessor, feature_names,
                   metadata, save_dir='./models'):
        """
        保存模型和相关文件
        """
        import joblib
        import json
        from datetime import datetime
        import os

        os.makedirs(save_dir, exist_ok=True)

        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        version = metadata.get('version', '1.0.0')

        # 1. 保存模型
        model_path = f'{save_dir}/model_{version}_{timestamp}.pkl'
        joblib.dump(model, model_path)
        print(f"✅ 模型已保存: {model_path}")

        # 2. 保存预处理器
        preprocessor_path = f'{save_dir}/preprocessor_{version}_{timestamp}.pkl'
        joblib.dump(preprocessor, preprocessor_path)
        print(f"✅ 预处理器已保存: {preprocessor_path}")

        # 3. 保存特征名
        features_path = f'{save_dir}/features_{version}_{timestamp}.json'
        with open(features_path, 'w') as f:
            json.dump({'feature_names': feature_names}, f, indent=2)
        print(f"✅ 特征信息已保存: {features_path}")

        # 4. 保存元数据
        metadata['timestamp'] = timestamp
        metadata['model_path'] = model_path
        metadata['preprocessor_path'] = preprocessor_path
        metadata['features_path'] = features_path

        metadata_path = f'{save_dir}/metadata_{version}_{timestamp}.json'
        with open(metadata_path, 'w') as f:
            json.dump(metadata, f, indent=2)
        print(f"✅ 元数据已保存: {metadata_path}")

        return {
            'model_path': model_path,
            'preprocessor_path': preprocessor_path,
            'features_path': features_path,
            'metadata_path': metadata_path,
        }

    def load_model(self, model_path, preprocessor_path):
        """
        加载模型
        """
        import joblib

        model = joblib.load(model_path)
        preprocessor = joblib.load(preprocessor_path)

        print(f"✅ 模型已加载: {model_path}")
        print(f"✅ 预处理器已加载: {preprocessor_path}")

        return model, preprocessor

    def create_prediction_api(self, model, preprocessor):
        """
        创建预测API (伪代码示例)
        """

        def predict(input_data):
            """
            预测函数

            Args:
                input_data: dict or DataFrame

            Returns:
                prediction: 预测结果
                probability: 预测概率 (分类任务)
            """
            # 1. 预处理
            X = preprocessor.transform(input_data)

            # 2. 预测
            prediction = model.predict(X)

            # 3. 概率 (如果支持)
            probability = None
            if hasattr(model, 'predict_proba'):
                probability = model.predict_proba(X)

            return {
                'prediction': prediction.tolist(),
                'probability': probability.tolist() if probability is not None else None,
            }

        return predict
```

## 常见误区

### ❌ 误区1: 跳过数据探索

```python
"""
错误: 直接开始建模,不探索数据

后果:
- 不了解数据特点
- 可能有数据质量问题
- 模型效果差

正确:
- 先花时间理解数据
- 可视化分析
- 发现规律和问题
"""
```

### ❌ 误区2: 没有基线模型

```python
"""
错误: 直接训练复杂模型

问题:
- 不知道简单模型效果如何
- 无法评估复杂模型的提升
- 可能过度设计

正确:
- 先训练简单的基线模型
- 再尝试复杂模型
- 对比提升幅度
"""
```

### ❌ 误区3: 在测试集上调参

```python
"""
错误: 看到测试集结果不好,调整模型再测

后果:
- 过拟合到测试集
- 真实性能被高估
- 结果不可信

正确:
- 测试集只用一次!
- 所有调整在验证集上做
- 测试集只用于最终评估
"""
```

## 总结

### 核心流程

```
问题定义 → 数据收集 → 数据探索 → 数据预处理 →
特征工程 → 数据划分 → 模型训练 → 模型评估 →
诊断调优 → 最终评估 → 部署监控
```

### 关键原则

1. **循序渐进**: 从简单到复杂
2. **数据第一**: 好数据 > 好模型
3. **诊断驱动**: 根据诊断结果优化
4. **验证严格**: 测试集只用一次
5. **持续改进**: 部署后持续监控
