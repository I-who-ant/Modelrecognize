## 基本定义

**梯度下降(Gradient Descent)**: 通过迭代更新参数,沿着损失函数梯度的反方向移动,最小化损失函数的优化算法。

**神经网络梯度下降**: 结合前向传播和反向传播,更新所有层的权重和偏置参数。

**核心思想**:
- 前向传播: 计算预测值和损失
- 反向传播: 计算梯度
- 参数更新: 沿梯度反方向更新
- 迭代重复: 直到收敛

```python
import numpy as np

def sigmoid(z):
    """Sigmoid激活函数"""
    return 1 / (1 + np.exp(-z))

def relu(z):
    """ReLU激活函数"""
    return np.maximum(0, z)

# ============================================================
# 完整的神经网络梯度下降
# ============================================================

def train_neural_network(X, Y, layer_dims, learning_rate=0.01,
                         num_iterations=10000, print_cost=True):
    """
    神经网络训练的完整流程

    参数:
    X: (n_x, m) 训练数据
    Y: (1, m) 标签
    layer_dims: [n_x, n_h1, n_h2, ..., n_y] 各层神经元数
    learning_rate: 学习率
    num_iterations: 迭代次数

    流程:
    1. 初始化参数
    2. for 迭代次数:
         a. 前向传播
         b. 计算损失
         c. 反向传播
         d. 更新参数
    3. 返回参数
    """
    np.random.seed(1)

    # ========== 步骤1: 初始化参数 ==========
    parameters = initialize_parameters(layer_dims)
    costs = []

    # ========== 步骤2: 梯度下降循环 ==========
    for i in range(num_iterations):

        # ========== 步骤2a: 前向传播 ==========
        AL, caches = forward_propagation(X, parameters)

        # ========== 步骤2b: 计算损失 ==========
        cost = compute_cost(AL, Y)

        # ========== 步骤2c: 反向传播 ==========
        grads = backward_propagation(AL, Y, caches)

        # ========== 步骤2d: 更新参数 ==========
        parameters = update_parameters(parameters, grads, learning_rate)

        # 记录损失
        if i % 100 == 0:
            costs.append(cost)
            if print_cost:
                print(f"迭代 {i}: 损失 = {cost:.6f}")

    return parameters, costs


def initialize_parameters(layer_dims):
    """
    初始化多层网络参数

    layer_dims: [n[0], n[1], n[2], ..., n[L]]

    返回:
    parameters: {'W1': W1, 'b1': b1, 'W2': W2, 'b2': b2, ...}
    """
    np.random.seed(1)
    parameters = {}
    L = len(layer_dims)  # 层数(包括输入层)

    for l in range(1, L):
        # He初始化 (适合ReLU)
        parameters[f'W{l}'] = np.random.randn(
            layer_dims[l], layer_dims[l-1]
        ) * np.sqrt(2.0 / layer_dims[l-1])

        parameters[f'b{l}'] = np.zeros((layer_dims[l], 1))

        # 验证维度
        assert parameters[f'W{l}'].shape == (layer_dims[l], layer_dims[l-1])
        assert parameters[f'b{l}'].shape == (layer_dims[l], 1)

    return parameters


def forward_propagation(X, parameters):
    """
    前向传播 (多层)

    X: (n[0], m) 输入
    parameters: 参数字典

    返回:
    AL: (n[L], m) 最终输出
    caches: 中间值列表,供反向传播使用
    """
    caches = []
    A = X
    L = len(parameters) // 2  # 层数

    # 隐藏层 (ReLU)
    for l in range(1, L):
        A_prev = A
        W = parameters[f'W{l}']
        b = parameters[f'b{l}']

        Z = W @ A_prev + b
        A = relu(Z)

        # 保存中间值
        cache = (A_prev, W, b, Z)
        caches.append(cache)

    # 输出层 (Sigmoid)
    WL = parameters[f'W{L}']
    bL = parameters[f'b{L}']
    ZL = WL @ A + bL
    AL = sigmoid(ZL)

    cache = (A, WL, bL, ZL)
    caches.append(cache)

    return AL, caches


def compute_cost(AL, Y):
    """
    计算交叉熵损失

    AL: (1, m) 预测值
    Y: (1, m) 真实标签

    返回: cost (标量)
    """
    m = Y.shape[1]

    # 交叉熵损失
    cost = -np.sum(Y * np.log(AL) + (1 - Y) * np.log(1 - AL)) / m

    cost = np.squeeze(cost)  # 确保是标量
    assert cost.shape == ()

    return cost


def backward_propagation(AL, Y, caches):
    """
    反向传播 (多层)

    AL: (n[L], m) 最终输出
    Y: (1, m) 真实标签
    caches: 前向传播保存的中间值

    返回:
    grads: {'dW1': dW1, 'db1': db1, 'dW2': dW2, ...}
    """
    grads = {}
    L = len(caches)  # 层数
    m = AL.shape[1]
    Y = Y.reshape(AL.shape)

    # ========== 输出层梯度 (Sigmoid + 交叉熵) ==========

    # dL/dAL
    dAL = -(np.divide(Y, AL) - np.divide(1 - Y, 1 - AL))

    # 当前层cache
    current_cache = caches[L-1]
    A_prev, W, b, Z = current_cache

    # Sigmoid导数: g'(Z) = AL * (1 - AL)
    dZ = dAL * AL * (1 - AL)

    # 参数梯度
    grads[f'dW{L}'] = (1/m) * dZ @ A_prev.T
    grads[f'db{L}'] = (1/m) * np.sum(dZ, axis=1, keepdims=True)

    # 传递给上一层
    dA_prev = W.T @ dZ

    # ========== 隐藏层梯度 (ReLU) ==========

    for l in reversed(range(L-1)):
        current_cache = caches[l]
        A_prev, W, b, Z = current_cache

        # ReLU导数: g'(Z) = (Z > 0)
        dZ = dA_prev * (Z > 0)

        # 参数梯度
        grads[f'dW{l+1}'] = (1/m) * dZ @ A_prev.T
        grads[f'db{l+1}'] = (1/m) * np.sum(dZ, axis=1, keepdims=True)

        # 传递给上一层
        if l > 0:
            dA_prev = W.T @ dZ

    return grads


def update_parameters(parameters, grads, learning_rate):
    """
    梯度下降参数更新

    parameters: 当前参数
    grads: 梯度
    learning_rate: 学习率

    返回: 更新后的参数
    """
    L = len(parameters) // 2  # 层数

    for l in range(1, L + 1):
        # 梯度下降更新
        parameters[f'W{l}'] -= learning_rate * grads[f'dW{l}']
        parameters[f'b{l}'] -= learning_rate * grads[f'db{l}']

    return parameters


# ============================================================
# 示例: 训练2层神经网络
# ============================================================

print("=" * 60)
print("神经网络梯度下降示例")
print("=" * 60)

m = 400  # 样本数
n_x = 2  # 特征数
```

```python
# 1D损失函数示例 (单参数梯度下降)

def loss_1d(w):
    """
    简单的1D损失函数
    L(w) = (w - 2)² + 1
    最小值在 w = 2
    """
    return (w - 2)**2 + 1

def gradient_1d(w):
    """
    损失函数的梯度
    dL/dw = 2(w - 2)
    """
    return 2 * (w - 2)

# 梯度下降
w = 0.0  # 初始值
learning_rate = 0.1
num_iterations = 20

w_history = [w]
loss_history = [loss_1d(w)]

for i in range(num_iterations):
    # 计算梯度
    grad = gradient_1d(w)

    # 更新参数
    w = w - learning_rate * grad

    # 记录
    w_history.append(w)
    loss_history.append(loss_1d(w))

    if i % 5 == 0:
        print(f"迭代 {i}: w = {w:.4f}, L(w) = {loss_1d(w):.4f}, "
              f"梯度 = {grad:.4f}")

# 可视化
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 5))
```

```python
def generate_data(n_samples=1000):

b = 0.0

# ========== 1. Batch梯度下降 (使用全部数据) ==========

def batch_gradient_descent(X, Y, num_iterations=100):
    """
    Batch梯度下降: 每次使用全部m个样本

    优点: 稳定,梯度准确
    缺点: 慢(m很大时),内存占用大
    """
    w, b = 0.0, 0.0
    m = X.shape[1]
    costs = []

    for i in range(num_iterations):
        # 前向传播 (全部m个样本)
        Y_pred = w * X + b

        # 损失
        cost = np.mean((Y_pred - Y)**2)
        costs.append(cost)

        # 梯度 (全部m个样本)
        dw = (2/m) * np.sum((Y_pred - Y) * X)
        db = (2/m) * np.sum(Y_pred - Y)

        # 更新
        w -= learning_rate * dw
        b -= learning_rate * db

    return w, b, costs

# ========== 2. Mini-batch梯度下降 ==========

def mini_batch_gradient_descent(X, Y, batch_size=32, num_epochs=10):
    """
    Mini-batch梯度下降: 每次使用batch_size个样本

    优点: 平衡速度和稳定性
    缺点: 损失曲线有噪声
    """
    w, b = 0.0, 0.0
    m = X.shape[1]
    costs = []

    for epoch in range(num_epochs):
        # 打乱数据
        permutation = np.random.permutation(m)
        X_shuffled = X[:, permutation]
        Y_shuffled = Y[:, permutation]

        # 分批处理
        num_batches = m // batch_size

        for batch_idx in range(num_batches):
            start = batch_idx * batch_size
            end = start + batch_size

            X_batch = X_shuffled[:, start:end]
            Y_batch = Y_shuffled[:, start:end]

            # 前向传播
            Y_pred = w * X_batch + b

            # 损失
            cost = np.mean((Y_pred - Y_batch)**2)
            costs.append(cost)

            # 梯度
            batch_m = batch_size
            dw = (2/batch_m) * np.sum((Y_pred - Y_batch) * X_batch)
            db = (2/batch_m) * np.sum(Y_pred - Y_batch)

            # 更新
            w -= learning_rate * dw
            b -= learning_rate * db

    return w, b, costs

# ========== 3. SGD (随机梯度下降) ==========

def stochastic_gradient_descent(X, Y, num_epochs=10):
    """
    SGD: 每次使用1个样本

    优点: 快速,可以online学习
    缺点: 非常不稳定,损失曲线剧烈波动
    """
    w, b = 0.0, 0.0
    m = X.shape[1]
    costs = []

    for epoch in range(num_epochs):
        # 打乱数据
        permutation = np.random.permutation(m)

        for i in range(m):
            idx = permutation[i]
            x = X[:, idx:idx+1]
            y = Y[:, idx:idx+1]

            # 前向传播
            y_pred = w * x + b

            # 损失
            cost = (y_pred - y)**2
            costs.append(float(cost))

            # 梯度
            dw = 2 * (y_pred - y) * x
            db = 2 * (y_pred - y)

            # 更新
            w -= learning_rate * float(dw)
            b -= learning_rate * float(db)

    return w, b, costs

```

```
参数 = 你的位置
梯度 = 坡度(指向上坡)
梯度下降 = 沿着下坡方向走

过程:
1. 站在山上某位置(初始化参数)
2. 感受脚下的坡度(计算梯度)
3. 朝下坡方向迈一步(更新参数)
   - 学习率 = 步长
   - 学习率太小 = 步子太小,走得慢
   - 学习率太大 = 步子太大,可能越过山谷
4. 重复2-3,直到到达谷底(收敛)

Batch vs Mini-batch vs SGD:
- Batch: 看遍整个山(全部数据),然后迈一步
- Mini-batch: 看一小片山(一批数据),然后迈一步
- SGD: 看脚下一点(一个数据),就迈一步
```

```
梯度 = 杂音变化方向
目标 = 找到杂音最小的频率

过程:
```

### 1. AdamW优化器

```python
# DeepSeek-V3使用AdamW,是梯度下降的改进版

class AdamW:
    """
    AdamW = Adam + Weight Decay

    改进:
    1. 动量 (Momentum): 累积梯度方向
    2. 自适应学习率: 每个参数独立学习率
    3. 权重衰减: 防止过拟合
    """
    def __init__(self, lr=0.001, beta1=0.9, beta2=0.999,
                 eps=1e-8, weight_decay=0.01):
        self.lr = lr
        self.beta1 = beta1  # 一阶动量衰减率
        self.beta2 = beta2  # 二阶动量衰减率
        self.eps = eps
        self.weight_decay = weight_decay

        self.m = {}  # 一阶动量
        self.v = {}  # 二阶动量
        self.t = 0   # 时间步

    def update(self, params, grads):
        """AdamW更新"""
        self.t += 1

        for key in params:
            if key not in self.m:
                self.m[key] = np.zeros_like(params[key])
                self.v[key] = np.zeros_like(params[key])

            # 梯度
            g = grads[f'd{key}']

            # 更新一阶动量 (指数移动平均)
            self.m[key] = self.beta1 * self.m[key] + (1 - self.beta1) * g

            # 更新二阶动量
            self.v[key] = self.beta2 * self.v[key] + (1 - self.beta2) * (g**2)

            # 偏差修正
            m_hat = self.m[key] / (1 - self.beta1**self.t)
            v_hat = self.v[key] / (1 - self.beta2**self.t)

            # AdamW更新 (先权重衰减,再Adam更新)
            params[key] = params[key] * (1 - self.lr * self.weight_decay)  # 权重衰减
            params[key] -= self.lr * m_hat / (np.sqrt(v_hat) + self.eps)  # Adam更新

        return params

# 对比SGD和AdamW
"""
SGD:
  W = W - lr * dW
  - 简单
  - 所有参数用同一学习率
  - 可能在峡谷地形震荡

AdamW:
  W = W * (1 - lr * λ) - lr * m / (√v + ε)
  - 自适应学习率
  - 每个参数独立
  - 收敛更快更稳定
"""
```

```python
# DeepSeek-V3的学习率调度策略

def cosine_annealing_with_warmup(t, T_warmup, T_max, lr_max, lr_min=0):
    """
    余弦退火 + Warmup

    DeepSeek-V3使用类似策略:
    1. Warmup阶段: 线性增加学习率
    2. 余弦退火: 学习率按余弦曲线下降

    参数:
    t: 当前step
    T_warmup: warmup步数
    T_max: 总步数
    lr_max: 最大学习率
    lr_min: 最小学习率
    """
    if t < T_warmup:
        # Warmup: 线性增加
        return lr_max * (t / T_warmup)
    else:
        # 余弦退火
        t_adjusted = t - T_warmup
        T_adjusted = T_max - T_warmup
        return lr_min + 0.5 * (lr_max - lr_min) * \
               (1 + np.cos(np.pi * t_adjusted / T_adjusted))
```

```python
# 学习率查找方法 (Learning Rate Finder)

def find_learning_rate(model, X, Y, lr_min=1e-6, lr_max=10, num_iter=100):
    """
    学习率查找器

    方法:
    1. 从极小学习率开始
    2. 每步指数增加学习率
    3. 记录损失
    4. 找损失下降最快的学习率
    """
    lrs = []
    losses = []

    # 初始化
    parameters = initialize_parameters([X.shape[0], 4, Y.shape[0]])

    # 指数增长的学习率
    lr_mult = (lr_max / lr_min) ** (1 / num_iter)
    lr = lr_min

    for i in range(num_iter):
        # 前向传播
        AL, caches = forward_propagation(X, parameters)

        # 损失
        cost = compute_cost(AL, Y)

        # 反向传播
        grads = backward_propagation(AL, Y, caches)

        # 更新
        parameters = update_parameters(parameters, grads, lr)

        # 记录
        lrs.append(lr)
        losses.append(cost)

        # 增加学习率
        lr *= lr_mult

        # 如果损失爆炸,停止
        if cost > losses[0] * 4:
            break

    # 可视化
    plt.figure(figsize=(10, 5))
    plt.plot(lrs, losses, 'b-', linewidth=2)
    plt.xscale('log')
    plt.xlabel('学习率')
    plt.ylabel('损失')
    plt.title('学习率查找')
    plt.grid(True, alpha=0.3)

    # 找最优学习率 (损失下降最快的点)
    gradients = np.diff(losses)
    min_idx = np.argmin(gradients)
    optimal_lr = lrs[min_idx]

    plt.axvline(x=optimal_lr, color='r', linestyle='--', linewidth=2,
                label=f'建议lr={optimal_lr:.6f}')
    plt.legend()
    plt.savefig('learning_rate_finder.png', dpi=100, bbox_inches='tight')
    plt.show()

    return optimal_lr

# 经验法则
"""
1. 从1e-3或1e-4开始尝试
2. 如果损失不下降 → 学习率太小
3. 如果损失爆炸/NaN → 学习率太大
4. 最优学习率通常在损失下降最快的点
5. 实际使用时可以比最优值小一个数量级(更稳定)
"""
```

```python
# 训练终止条件

def should_stop_training(costs, patience=10, min_delta=1e-4):
    """
    判断是否应该停止训练

    条件:
    1. 损失不再下降 (Early Stopping)
    2. 达到最大迭代次数
    3. 损失低于阈值
    4. 验证集性能不再提升
    """
    # 1. Early Stopping
    if len(costs) > patience:
        recent_costs = costs[-patience:]
        if max(recent_costs) - min(recent_costs) < min_delta:
            return True, "损失不再下降"

    # 2. 达到目标
    if costs[-1] < 0.01:
        return True, "达到目标损失"

    return False, None
```

```python
   - Batch size ↑ → 梯度准确 ↑ → 稳定但可能慢

   - 大batch size: 可能过拟合到sharp minima

1. 从小开始 (如32)
2. 逐步增大,直到GPU满载或OOM
3. 记录训练时间和最终性能
4. 选择性价比最高的
```

## 总结

### 核心公式

```python
# 梯度下降更新规则
W = W - α · ∂L/∂W
b = b - α · ∂L/∂b

其中:
- W, b: 参数
- α: 学习率 (learning rate)
- ∂L/∂W, ∂L/∂b: 梯度

# 完整训练循环
for iteration in range(num_iterations):
    # 1. 前向传播
    AL, caches = forward_propagation(X, parameters)

    # 2. 计算损失
    cost = compute_cost(AL, Y)

    # 3. 反向传播
    grads = backward_propagation(AL, Y, caches)

    # 4. 更新参数
    parameters = update_parameters(parameters, grads, learning_rate)
```

### 关键记忆点

1. **梯度下降 = 迭代优化**: 前向→损失→反向→更新
2. **学习率很关键**: 太小收敛慢,太大不稳定
3. **三种变体**: Batch, Mini-batch, SGD
4. **实际用Mini-batch**: 平衡速度和稳定性
5. **监控损失曲线**: 诊断训练问题
6. **Early Stopping**: 防止过拟合
