### 2.1 准确性指标

```python
import json
from collections import Counter
import re

class AccuracyEvaluator:
    """准确性评估器"""

    def __init__(self):
        self.errors = []
        self.error_types = Counter()

    def evaluate_qa_accuracy(self, qa_dataset: list) -> dict:
        """
        评估问答数据准确性

        返回:
            {
                'accuracy': 0.92,
                'error_types': {'factual': 5, 'logic': 2},
                'error_examples': [...]
            }
        """
        correct = 0
        errors = []

        for item in qa_dataset:
            question = item.get('question', '')
            answer = item.get('answer', '')

            # 检查1: 答案长度合理(避免过短)
            if len(answer) < 20:
                errors.append({
                    'question': question,
                    'error': '答案过短',
                    'type': 'length'
                })
                continue

            # 检查2: 逻辑一致性
            if self._check_logic_consistency(question, answer):
                # 检查3: 是否包含矛盾表述
                if not self._check_contradiction(answer):
                    correct += 1
                else:
                    errors.append({
                        'question': question,
                        'answer': answer,
                        'error': '自相矛盾',
                        'type': 'contradiction'
                    })
            else:
                errors.append({
                    'question': question,
                    'answer': answer,
                    'error': '问答不匹配',
                    'type': 'mismatch'
                })

        accuracy = correct / len(qa_dataset) if qa_dataset else 0

        return {
            'accuracy': accuracy,
            'correct_count': correct,
            'error_count': len(errors),
            'error_types': dict(Counter([e['type'] for e in errors])),
            'error_examples': errors[:10]  # 返回前10个错误示例
        }

    def _check_logic_consistency(self, question: str, answer: str) -> bool:
        """检查逻辑一致性"""
        # 简单检查: 答案中是否包含问题的关键词
        q_words = set(question.lower().split())
        a_words = set(answer.lower().split())

        # 至少有30%的关键词在答案中出现
        overlap = len(q_words & a_words) / len(q_words) if q_words else 0
        return overlap > 0.3

    def _check_contradiction(self, text: str) -> bool:
        """检查自相矛盾"""
        # 检查是否包含矛盾表述(简单规则)
        contradictions = [
            (r'不是', r'是'),
            (r'没有', r'有'),
            (r'不会', r'会'),
        ]

        for neg, pos in contradictions:
            if re.search(neg, text) and re.search(pos, text):
                return True  # 有矛盾
        return False
```

### 2.2 多样性指标

```python
import math
from collections import Counter

class DiversityEvaluator:
    """多样性评估器"""

    def evaluate_diversity(self, dataset: list) -> dict:
        """
        评估数据集多样性
        """
        # 提取所有文本
        texts = [self._get_text(item) for item in dataset]

        # 1. 词汇多样性 (Type/Token Ratio)
        ttr = self._calculate_ttr(texts)

        # 2. 独特词汇数
        vocab_size = self._calculate_vocab_size(texts)

        # 3. 信息熵
        entropy = self._calculate_entropy(texts)

        # 4. 长度分布多样性
        length_std = self._calculate_length_std(texts)

        # 5. 场景覆盖多样性(如果有场景标签)
        scene_diversity = self._calculate_scene_diversity(dataset)

        return {
            'type_token_ratio': ttr,
            'vocab_size': vocab_size,
            'entropy': entropy,
            'length_std': length_std,
            'scene_diversity': scene_diversity,
            'diversity_score': self._calculate_overall_score(ttr, entropy, vocab_size)
        }

    def _calculate_ttr(self, texts: list) -> float:
        """
        计算Type/Token Ratio
        越高越多样化 (目标: > 0.4)
        """
        all_words = []
        for text in texts:
            words = text.lower().split()
            all_words.extend(words)

        unique_words = set(all_words)
        ttr = len(unique_words) / len(all_words) if all_words else 0
        return ttr

    def _calculate_vocab_size(self, texts: list) -> int:
        """计算词汇表大小"""
        vocab = set()
        for text in texts:
            words = text.lower().split()
            vocab.update(words)
        return len(vocab)

    def _calculate_entropy(self, texts: list) -> float:
        """
        计算信息熵
        衡量数据的随机性和多样性
        越高越多样化 (目标: > 3.5)
        """
        word_freq = Counter()
        total_words = 0

        for text in texts:
            words = text.lower().split()
            word_freq.update(words)
            total_words += len(words)

        # 计算香农熵: 衡量词汇分布的随机性
        entropy = 0
        for count in word_freq.values():
            p = count / total_words
            if p > 0:
                entropy -= p * math.log2(p)

        return entropy

    def _calculate_length_std(self, texts: list) -> float:
        """
        计算文本长度的标准差
        衡量长度的多样性
        """
        lengths = [len(text.split()) for text in texts]
        mean = sum(lengths) / len(lengths) if lengths else 0
        variance = sum((l - mean) ** 2 for l in lengths) / len(lengths) if lengths else 0
        std = math.sqrt(variance)
        return std

    def _calculate_scene_diversity(self, dataset: list) -> float:
        """
        计算场景覆盖多样性
        如果数据有'scene'字段,统计各场景分布
        """
        if not any('scene' in item for item in dataset):
            return None

        scenes = Counter(item.get('scene', 'unknown') for item in dataset)
        total = len(dataset)

        # 使用信息熵衡量分布
        entropy = 0
        for count in scenes.values():
            p = count / total
            if p > 0:
                entropy -= p * math.log2(p)

        return entropy

    def _calculate_overall_score(self, ttr: float, entropy: float, vocab_size: int) -> float:
        """
        计算综合多样性评分 (0-10)
        """
        # 归一化各指标
        ttr_score = min(ttr / 0.4, 1.0) * 3  # 满分3分
        entropy_score = min(entropy / 4.0, 1.0) * 4  # 满分4分
        vocab_score = min(vocab_size / 10000, 1.0) * 3  # 满分3分

        overall = ttr_score + entropy_score + vocab_score
        return min(overall, 10.0)

    def _get_text(self, item: dict) -> str:
        """提取数据项的主要文本"""
        if 'text' in item:
            return item['text']
        elif 'content' in item:
            return item['content']
        elif 'answer' in item:
            return item['answer']
        return ''
```

### 2.3 真实性指标

```python
import numpy as np
from scipy.spatial.distance import wasserstein_distance

class RealismEvaluator:
    """真实性评估器"""

    def evaluate_realism(self, synthetic_dataset: list, real_dataset: list = None) -> dict:
        """
        评估合成数据的真实性

        参数:
            synthetic_dataset: 合成数据
            real_dataset: 真实数据(可选,用于对比)

        返回:
            {
                'length_distribution_match': 0.92,
                'vocabulary_similarity': 0.85,
                'overall_realism_score': 0.88,
                'distribution_analysis': {...}
            }
        """
        results = {}

        # 1. 长度分布对比
        if real_dataset:
            synthetic_lengths = [len(self._get_text(item).split()) for item in synthetic_dataset]
            real_lengths = [len(self._get_text(item).split()) for item in real_dataset]

            # 使用Wasserstein距离衡量分布差异
            wasserstein = wasserstein_distance(synthetic_lengths, real_lengths)
            results['length_distribution_match'] = 1.0 / (1.0 + wasserstein)

            # KL散度
            results['kl_divergence'] = self._calculate_kl_divergence(synthetic_lengths, real_lengths)

        # 2. 词汇相似性
        results['vocabulary_overlap'] = self._calculate_vocab_overlap(
            synthetic_dataset, real_dataset
        ) if real_dataset else None

        # 3. 合成数据整体真实性评分
        results['synthetic_realism_score'] = self._evaluate_synthetic_realism(synthetic_dataset)

        # 4. 分布分析
        results['distribution_analysis'] = self._analyze_distribution(synthetic_dataset)

        # 计算综合评分
        scores = [v for v in results.values() if isinstance(v, float) and 0 <= v <= 1]
        results['overall_realism_score'] = np.mean(scores) if scores else 0.5

        return results

    def _calculate_kl_divergence(self, synthetic: list, real: list) -> float:
        """
        计算KL散度
        衡量两个分布的差异 (0最接近)
        """
        # 创建直方图
        all_values = synthetic + real
        min_val, max_val = min(all_values), max(all_values)
        bins = np.linspace(min_val, max_val, 20)

        p, _ = np.histogram(synthetic, bins=bins)
        q, _ = np.histogram(real, bins=bins)

        # 归一化
        p = (p + 1e-10) / (np.sum(p) + 1e-9)
        q = (q + 1e-10) / (np.sum(q) + 1e-9)

        # 计算KL散度
        kl = np.sum(p * np.log(p / q))
        return float(kl)

    def _calculate_vocab_overlap(self, synthetic_dataset: list, real_dataset: list) -> float:
        """计算词汇重叠度"""
        synthetic_vocab = set()
        for item in synthetic_dataset:
            words = self._get_text(item).lower().split()
            synthetic_vocab.update(words)

        real_vocab = set()
        for item in real_dataset:
            words = self._get_text(item).lower().split()
            real_vocab.update(words)

        # Jaccard相似度
        overlap = len(synthetic_vocab & real_vocab)
        union = len(synthetic_vocab | real_vocab)
        return overlap / union if union > 0 else 0

    def _evaluate_synthetic_realism(self, dataset: list) -> float:
        """
        评估合成数据的真实性评分

        检查项:
        1. 是否包含常见的语言模式
        2. 是否避免了生成陷阱(如过度完美)
        3. 是否包含合理的多样性
        """
        realism_score = 0.8  # 初始分数

        # 检查1: 是否过于完美(避免这是信号)
        texts = [self._get_text(item) for item in dataset]
        avg_length = np.mean([len(t.split()) for t in texts])

        # 如果文本过短或过长,降分
        if avg_length < 10 or avg_length > 200:
            realism_score -= 0.1

        # 检查2: 是否有自然的多样性
        unique_texts = len(set(texts))
        diversity_ratio = unique_texts / len(texts)

        if diversity_ratio < 0.7:  # 如果重复率过高
            realism_score -= 0.15

        # 检查3: 是否包含常见的表达模式
        common_patterns = ['但是', '而且', '因此', '例如', '总之']
        texts_str = ' '.join(texts)
        pattern_coverage = sum(1 for p in common_patterns if p in texts_str) / len(common_patterns)

        if pattern_coverage < 0.5:
            realism_score -= 0.1

        return max(realism_score, 0.4)

    def _analyze_distribution(self, dataset: list) -> dict:
        """分析数据分布特征"""
        texts = [self._get_text(item) for item in dataset]
        lengths = [len(t.split()) for t in texts]

        return {
            'avg_length': np.mean(lengths),
            'min_length': np.min(lengths),
            'max_length': np.max(lengths),
            'std_length': np.std(lengths),
            'median_length': np.median(lengths),
            'text_count': len(texts),
            'unique_text_count': len(set(texts))
        }

    def _get_text(self, item: dict) -> str:
        """提取文本"""
        for key in ['text', 'content', 'answer', 'question']:
            if key in item:
                return item[key]
        return ''
```

### 3.1 合成数据 vs 真实数据对比

```python
class ComparativeAnalyzer:
    """对比分析器"""

    def compare_datasets(self, synthetic: list, real: list) -> dict:
        """
        全面对比合成数据和真实数据

        返回详细的对比报告
        """
        comparison = {}

        # 1. 基本统计对比
        comparison['statistics'] = self._compare_statistics(synthetic, real)

        # 2. 分布对比
        comparison['distribution'] = self._compare_distribution(synthetic, real)

        # 3. 质量指标对比
        comparison['quality_metrics'] = self._compare_quality_metrics(synthetic, real)

        # 4. 词汇对比
        comparison['vocabulary'] = self._compare_vocabulary(synthetic, real)

        # 5. 综合评分
        comparison['overall_score'] = self._calculate_overall_comparison_score(comparison)

        return comparison

    def _compare_statistics(self, synthetic: list, real: list) -> dict:
        """比较基本统计特征"""
        import numpy as np

        def get_stats(dataset):
            texts = [self._get_text(item) for item in dataset]
            lengths = [len(t.split()) for t in texts]
            return {
                'count': len(texts),
                'avg_length': np.mean(lengths),
                'std_length': np.std(lengths),
                'min_length': np.min(lengths),
                'max_length': np.max(lengths)
            }

        synthetic_stats = get_stats(synthetic)
        real_stats = get_stats(real)

        # 计算差异
        diff = {}
        for key in synthetic_stats:
            if key == 'count':
                continue
            s_val = synthetic_stats[key]
            r_val = real_stats[key]
            # 相对差异百分比
            diff[key] = abs(s_val - r_val) / (r_val + 1e-6) * 100

        return {
            'synthetic': synthetic_stats,
            'real': real_stats,
            'difference_percent': diff
        }

    def _compare_distribution(self, synthetic: list, real: list) -> dict:
        """比较分布特征"""
        import numpy as np

        synthetic_lengths = [len(self._get_text(item).split()) for item in synthetic]
        real_lengths = [len(self._get_text(item).split()) for item in real]

        # 计算各种距离
        from scipy.spatial.distance import wasserstein_distance, ks_2samp

        wasserstein_dist = wasserstein_distance(synthetic_lengths, real_lengths)
        ks_stat, ks_pvalue = ks_2samp(synthetic_lengths, real_lengths)

        return {
            'wasserstein_distance': wasserstein_dist,
            'ks_statistic': ks_stat,
            'ks_pvalue': ks_pvalue,
            'ks_match': 'Yes' if ks_pvalue > 0.05 else 'No'  # p>0.05表示分布相同
        }

    def _compare_quality_metrics(self, synthetic: list, real: list) -> dict:
        """比较质量指标"""
        # 使用之前定义的评估器
        acc_eval = AccuracyEvaluator()
        div_eval = DiversityEvaluator()

        synthetic_acc = acc_eval.evaluate_qa_accuracy(synthetic)
        real_acc = acc_eval.evaluate_qa_accuracy(real)

        synthetic_div = div_eval.evaluate_diversity(synthetic)
        real_div = div_eval.evaluate_diversity(real)

        return {
            'accuracy': {
                'synthetic': synthetic_acc['accuracy'],
                'real': real_acc['accuracy'],
                'gap': abs(synthetic_acc['accuracy'] - real_acc['accuracy'])
            },
            'diversity': {
                'synthetic': synthetic_div['diversity_score'],
                'real': real_div['diversity_score'],
                'gap': abs(synthetic_div['diversity_score'] - real_div['diversity_score'])
            }
        }

    def _compare_vocabulary(self, synthetic: list, real: list) -> dict:
        """比较词汇特征"""
        def get_vocab(dataset):
            vocab = set()
            for item in dataset:
                words = self._get_text(item).lower().split()
                vocab.update(words)
            return vocab

        synthetic_vocab = get_vocab(synthetic)
        real_vocab = get_vocab(real)

        overlap = synthetic_vocab & real_vocab
        unique_to_synthetic = synthetic_vocab - real_vocab
        unique_to_real = real_vocab - synthetic_vocab

        return {
            'synthetic_vocab_size': len(synthetic_vocab),
            'real_vocab_size': len(real_vocab),
            'overlap': len(overlap),
            'overlap_ratio': len(overlap) / len(real_vocab) if real_vocab else 0,
            'unique_to_synthetic': len(unique_to_synthetic),
            'unique_to_real': len(unique_to_real),
            'jaccard_similarity': len(overlap) / len(synthetic_vocab | real_vocab)
        }

    def _calculate_overall_comparison_score(self, comparison: dict) -> float:
        """计算综合对比评分"""
        # 各项权重
        weights = {
            'distribution': 0.3,
            'quality': 0.4,
            'vocabulary': 0.3
        }

        score = 0

        # 分布评分 (KS test p-value越大越好)
        if comparison['distribution']['ks_pvalue'] > 0.05:
            dist_score = 1.0
        else:
            dist_score = comparison['distribution']['ks_pvalue'] / 0.05
        score += dist_score * weights['distribution']

        # 质量评分
        qual_score = 1.0 - (
            comparison['quality_metrics']['accuracy']['gap'] +
            comparison['quality_metrics']['diversity']['gap']
        ) / 2
        score += max(qual_score, 0) * weights['quality']

        # 词汇评分
        vocab_score = comparison['vocabulary']['jaccard_similarity']
        score += vocab_score * weights['vocabulary']

        return score
```

## 📚 第四部分: 完整评估管道

### 4.1 端到端评估系统

```python
class ComprehensiveEvaluator:
    """综合评估器"""

    def __init__(self, config: dict = None):
        self.config = config or {}
        self.evaluators = {
            'accuracy': AccuracyEvaluator(),
            'diversity': DiversityEvaluator(),
            'realism': RealismEvaluator(),
            'comparison': ComparativeAnalyzer()
        }

    def evaluate_full_pipeline(
        self,
        synthetic_dataset: list,
        real_dataset: list = None,
        output_report: str = None
    ) -> dict:
        """
        执行完整评估管道

        返回综合评估报告
        """
        print("开始数据集评估...")

        # 第1步: 准确性评估
        print("\n[1/4] 评估准确性...")
        accuracy_report = self.evaluators['accuracy'].evaluate_qa_accuracy(synthetic_dataset)

        # 第2步: 多样性评估
        print("[2/4] 评估多样性...")
        diversity_report = self.evaluators['diversity'].evaluate_diversity(synthetic_dataset)

        # 第3步: 真实性评估
        print("[3/4] 评估真实性...")
        realism_report = self.evaluators['realism'].evaluate_realism(
            synthetic_dataset, real_dataset
        )

        # 第4步: 对比分析(如果有真实数据)
        comparison_report = None
        if real_dataset:
            print("[4/4] 执行对比分析...")
            comparison_report = self.evaluators['comparison'].compare_datasets(
                synthetic_dataset, real_dataset
            )
        else:
            print("[4/4] 跳过对比分析(无真实数据参考)")

        # 综合报告
        report = {
            'timestamp': self._get_timestamp(),
            'dataset_size': len(synthetic_dataset),
            'accuracy': accuracy_report,
            'diversity': diversity_report,
            'realism': realism_report,
            'comparison': comparison_report,
            'overall_quality_score': self._calculate_overall_score(
                accuracy_report, diversity_report, realism_report, comparison_report
            ),
            'recommendations': self._generate_recommendations(
                accuracy_report, diversity_report, realism_report
            )
        }

        # 打印报告
        self._print_report(report)

        # 保存报告
        if output_report:
            self._save_report(report, output_report)

        return report

    def _calculate_overall_score(self, acc, div, real, comp) -> float:
        """计算综合质量评分 (0-100分)"""
        # 权重配置
        weights = {
            'accuracy': 0.25,
            'diversity': 0.20,
            'realism': 0.25,
            'comparison': 0.30
        }

        score = 0

        # 准确性评分
        acc_score = min(acc['accuracy'] * 100, 100)
        score += acc_score * weights['accuracy']

        # 多样性评分
        div_score = min(div['diversity_score'] * 10, 100)
        score += div_score * weights['diversity']

        # 真实性评分
        real_score = min(real['overall_realism_score'] * 100, 100)
        score += real_score * weights['realism']

        # 对比评分
        if comp:
            comp_score = min(comp['overall_score'] * 100, 100)
            score += comp_score * weights['comparison']
        else:
            score += 75 * weights['comparison']  # 无比较数据时给默认分

        return score

    def _generate_recommendations(self, acc, div, real) -> list:
        """基于评估结果生成改进建议"""
        recommendations = []

        # 准确性相关建议
        if acc['accuracy'] < 0.85:
            recommendations.append({
                'issue': '准确性不足',
                'current': f"{acc['accuracy']:.1%}",
                'target': '95%+',
                'suggestions': [
                    '使用更强的模型(如GPT-4)',
                    '改进生成提示词,明确要求准确性',
                    '增加人工审核比例'
                ]
            })

        # 多样性相关建议
        if div['diversity_score'] < 6.0:
            recommendations.append({
                'issue': '多样性不足',
                'current': f"{div['diversity_score']:.1f}/10",
                'target': '8.0+',
                'suggestions': [
                    '提高temperature参数(0.8-1.0)',
                    '使用多个提示词变体',
                    '增加生成次数,避免重复'
                ]
            })

        # 真实性相关建议
        if real['overall_realism_score'] < 0.75:
            recommendations.append({
                'issue': '真实性不足',
                'current': f"{real['overall_realism_score']:.1%}",
                'target': '85%+',
                'suggestions': [
                    '在生成提示词中明确要求自然表达',
                    '添加常见的语言模式和表达方式',
                    '参考真实数据调整生成策略'
                ]
            })

        return recommendations

    def _print_report(self, report: dict):
        """打印评估报告"""
        print("\n" + "="*60)
        print("📊 数据集质量评估报告")
        print("="*60)

        print(f"\n📈 综合质量评分: {report['overall_quality_score']:.1f}/100")
        print(f"   数据集大小: {report['dataset_size']}条")

        print(f"\n📋 准确性评估:")
        print(f"   - 准确率: {report['accuracy']['accuracy']:.1%}")
        print(f"   - 错误数: {report['accuracy']['error_count']}条")
        if report['accuracy']['error_types']:
            print(f"   - 错误类型: {report['accuracy']['error_types']}")

        print(f"\n🎯 多样性评估:")
        print(f"   - Type/Token Ratio: {report['diversity']['type_token_ratio']:.3f}")
        print(f"   - 词汇表大小: {report['diversity']['vocab_size']}")
        print(f"   - 信息熵: {report['diversity']['entropy']:.3f}")
        print(f"   - 综合评分: {report['diversity']['diversity_score']:.1f}/10")

        print(f"\n✨ 真实性评估:")
        print(f"   - 真实性评分: {report['realism']['overall_realism_score']:.1%}")

        if report['comparison']:
            print(f"\n🔄 对比分析:")
            print(f"   - KS检验 p值: {report['comparison']['distribution']['ks_pvalue']:.4f}")
            print(f"   - Jaccard相似度: {report['comparison']['vocabulary']['jaccard_similarity']:.3f}")
            print(f"   - 综合对比评分: {report['comparison']['overall_score']:.3f}/1.0")

        if report['recommendations']:
            print(f"\n💡 改进建议:")
            for i, rec in enumerate(report['recommendations'], 1):
                print(f"\n   {i}. {rec['issue']}")
                print(f"      - 当前: {rec['current']}")
                print(f"      - 目标: {rec['target']}")
                print(f"      - 建议:")
                for suggestion in rec['suggestions']:
                    print(f"        • {suggestion}")

        print("\n" + "="*60 + "\n")

    def _save_report(self, report: dict, path: str):
        """保存评估报告"""
        import json
        with open(path, 'w', encoding='utf-8') as f:
            json.dump(report, f, ensure_ascii=False, indent=2)
        print(f"✅ 报告已保存: {path}")

    def _get_timestamp(self) -> str:
        """获取时间戳"""
        from datetime import datetime
        return datetime.now().strftime('%Y-%m-%d %H:%M:%S')
```
