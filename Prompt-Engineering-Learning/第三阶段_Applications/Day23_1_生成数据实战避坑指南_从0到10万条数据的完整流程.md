```python
# src/generator.py
import openai
import time
import logging
from typing import List, Dict
from tenacity import retry, stop_after_attempt, wait_exponential
from tqdm import tqdm

class DataGenerator:
    """数据生成器基类"""

    def __init__(self, config: Dict):
        self.config = config
        self.model = config['generation']['model']
        self.temperature = config['generation']['temperature']
        self.max_tokens = config['generation']['max_tokens']

        # 设置日志
        logging.basicConfig(
            filename='logs/generation.log',
            level=logging.INFO,
            format='%(asctime)s - %(levelname)s - %(message)s'
        )
        self.logger = logging.getLogger(__name__)

        # 统计信息
        self.stats = {
            'total_generated': 0,
            'total_failed': 0,
            'total_cost': 0.0,
            'start_time': None,
            'end_time': None
        }

    @retry(
        stop=stop_after_attempt(3),
        wait=wait_exponential(multiplier=1, min=4, max=10)
    )
    def _call_llm(self, prompt: str) -> str:
        """
        调用LLM API (带重试机制)
        """
        try:
            response = openai.ChatCompletion.create(
                model=self.model,
                messages=[{"role": "user", "content": prompt}],
                temperature=self.temperature,
                max_tokens=self.max_tokens
            )

            # 统计token和成本
            usage = response.usage
            cost = self._calculate_cost(usage.prompt_tokens, usage.completion_tokens)
            self.stats['total_cost'] += cost

            return response.choices[0].message.content

        except openai.error.RateLimitError:
            self.logger.warning("Rate limit hit, waiting...")
            time.sleep(60)  # 等待1分钟
            raise  # 重试

        except openai.error.APIError as e:
            self.logger.error(f"API Error: {e}")
            raise

        except Exception as e:
            self.logger.error(f"Unexpected error: {e}")
            self.stats['total_failed'] += 1
            raise

    def _calculate_cost(self, input_tokens: int, output_tokens: int) -> float:
        """计算成本"""
        if self.model == "gpt-4":
            input_cost = input_tokens / 1000 * self.config['costs']['gpt4_input']
            output_cost = output_tokens / 1000 * self.config['costs']['gpt4_output']
        else:  # gpt-3.5-turbo
            input_cost = input_tokens / 1000 * self.config['costs']['gpt35_input']
            output_cost = output_tokens / 1000 * self.config['costs']['gpt35_output']

        return input_cost + output_cost

    def generate_batch(self, prompt: str, batch_size: int = 10) -> List[Dict]:
        """
        生成一批数据

        参数:
            prompt: 生成提示词
            batch_size: 批次大小

        返回:
            生成的数据列表
        """
        # 修改提示词,要求生成指定数量
        batch_prompt = f"{prompt}\n\n请生成{batch_size}条数据。"

        # 调用LLM
        response = self._call_llm(batch_prompt)

        # 解析结果
        try:
            data = self._parse_response(response, batch_size)
            self.stats['total_generated'] += len(data)
            return data
        except Exception as e:
            self.logger.error(f"Parse error: {e}")
            self.stats['total_failed'] += batch_size
            return []

    def _parse_response(self, response: str, expected_count: int) -> List[Dict]:
        """
        解析LLM响应

        需要根据具体任务实现解析逻辑
        """
        raise NotImplementedError("Subclass must implement _parse_response")

    def generate_dataset(
        self,
        prompt: str,
        total_samples: int,
        batch_size: int = 10,
        save_path: str = None
    ) -> List[Dict]:
        """
        生成完整数据集

        参数:
            prompt: 生成提示词
            total_samples: 总样本数
            batch_size: 批次大小
            save_path: 保存路径

        返回:
            完整数据集
        """
        self.stats['start_time'] = time.time()
        self.logger.info(f"开始生成{total_samples}条数据...")

        dataset = []
        num_batches = (total_samples + batch_size - 1) // batch_size

        # 进度条
        with tqdm(total=num_batches, desc="生成进度") as pbar:
            for i in range(num_batches):
                # 最后一批可能不足batch_size
                current_batch_size = min(batch_size, total_samples - len(dataset))

                # 生成
                batch_data = self.generate_batch(prompt, current_batch_size)
                dataset.extend(batch_data)

                # 更新进度
                pbar.update(1)
                pbar.set_postfix({
                    'generated': len(dataset),
                    'failed': self.stats['total_failed'],
                    'cost': f"${self.stats['total_cost']:.2f}"
                })

                # 速率限制
                time.sleep(60 / self.config['generation']['rate_limit'])

                # 定期保存(每1000条)
                if save_path and len(dataset) % 1000 == 0:
                    self._save_checkpoint(dataset, save_path, i)

        self.stats['end_time'] = time.time()
        self._print_summary()

        # 最终保存
        if save_path:
            self._save_final(dataset, save_path)

        return dataset

    def _save_checkpoint(self, dataset: List[Dict], path: str, batch_num: int):
        """保存检查点"""
        import json
        checkpoint_path = f"{path}.checkpoint_{batch_num}.json"
        with open(checkpoint_path, 'w', encoding='utf-8') as f:
            json.dump(dataset, f, ensure_ascii=False, indent=2)
        self.logger.info(f"检查点已保存: {checkpoint_path}")

    def _save_final(self, dataset: List[Dict], path: str):
        """保存最终数据集"""
        import json
        with open(path, 'w', encoding='utf-8') as f:
            json.dump(dataset, f, ensure_ascii=False, indent=2)
        self.logger.info(f"数据集已保存: {path}")

    def _print_summary(self):
        """打印统计摘要"""
        duration = self.stats['end_time'] - self.stats['start_time']
        print("\n" + "="*50)
        print("生成完成!")
        print("="*50)
        print(f"✅ 成功生成: {self.stats['total_generated']}条")
        print(f"❌ 失败: {self.stats['total_failed']}条")
        print(f"💰 总成本: ${self.stats['total_cost']:.2f}")
        print(f"⏱️  耗时: {duration/60:.1f}分钟")
        print(f"⚡ 速率: {self.stats['total_generated']/(duration/60):.1f}条/分钟")
        print("="*50)
```

```python
# 情感分析数据生成器
class SentimentDataGenerator(DataGenerator):
    """情感分析数据生成器"""

    def _parse_response(self, response: str, expected_count: int) -> List[Dict]:
        """解析情感分析数据"""
        import json

        # 假设LLM返回JSON数组格式
        try:
            # 尝试直接解析JSON
            data = json.loads(response)
            if isinstance(data, list):
                return data

        except json.JSONDecodeError:
            # JSON解析失败,尝试按行解析
            lines = response.strip().split('\n')
            data = []

            for line in lines:
                line = line.strip()
                if not line:
                    continue

                # 尝试解析单行JSON
                try:
                    item = json.loads(line)
                    if 'text' in item and 'label' in item:
                        data.append(item)
                except:
                    # 尝试解析格式: "文本内容" - 标签
                    if ' - ' in line:
                        parts = line.split(' - ')
                        if len(parts) == 2:
                            data.append({
                                'text': parts[0].strip('"'),
                                'label': parts[1].strip()
                            })

            return data

        return []

# 问答数据生成器
class QADataGenerator(DataGenerator):
    """问答数据生成器"""

    def _parse_response(self, response: str, expected_count: int) -> List[Dict]:
        """解析问答数据"""
        import json
        import re

        data = []

        # 方法1: 尝试JSON解析
        try:
            parsed = json.loads(response)
            if isinstance(parsed, list):
                return parsed
        except:
            pass

        # 方法2: 按标记解析 (Q: ... A: ...)
        qa_pattern = r'Q[:\s]+(.+?)\s*A[:\s]+(.+?)(?=Q:|$)'
        matches = re.findall(qa_pattern, response, re.DOTALL)

        for question, answer in matches:
            data.append({
                'question': question.strip(),
                'answer': answer.strip()
            })

        return data

# 对话数据生成器
class DialogueDataGenerator(DataGenerator):
    """多轮对话数据生成器"""

    def _parse_response(self, response: str, expected_count: int) -> List[Dict]:
        """解析对话数据"""
        import json

        # 按分隔符拆分对话组
        dialogues = response.split('---')
        data = []

        for dialogue_text in dialogues:
            dialogue_text = dialogue_text.strip()
            if not dialogue_text:
                continue

            # 解析对话轮次
            turns = []
            lines = dialogue_text.split('\n')

            for line in lines:
                line = line.strip()
                if not line:
                    continue

                # 解析 "角色: 内容" 格式
                if ':' in line:
                    role, content = line.split(':', 1)
                    role = role.strip()
                    content = content.strip()

                    # 标准化角色名
                    if role in ['客户', 'Customer', 'User']:
                        role = 'customer'
                    elif role in ['客服', 'Agent', 'Assistant']:
                        role = 'agent'

                    turns.append({
                        'role': role,
                        'content': content
                    })

            if turns:
                data.append({
                    'dialogue': turns
                })

        return data
```

```python
# src/parallel_generator.py
from concurrent.futures import ThreadPoolExecutor, as_completed
import threading

class ParallelDataGenerator(DataGenerator):
    """并行数据生成器"""

    def __init__(self, config: Dict):
        super().__init__(config)
        self.max_workers = config['generation']['max_workers']
        self.lock = threading.Lock()  # 线程锁保护统计数据

    def generate_dataset_parallel(
        self,
        prompt: str,
        total_samples: int,
        batch_size: int = 10,
        save_path: str = None
    ) -> List[Dict]:
        """
        并行生成数据集

        速度提升: max_workers倍
        """
        self.stats['start_time'] = time.time()
        self.logger.info(f"开始并行生成{total_samples}条数据 (workers={self.max_workers})...")

        dataset = []
        num_batches = (total_samples + batch_size - 1) // batch_size

        # 创建任务列表
        tasks = []
        for i in range(num_batches):
            current_batch_size = min(batch_size, total_samples - i * batch_size)
            tasks.append((prompt, current_batch_size))

        # 并行执行
        with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            # 提交所有任务
            futures = {
                executor.submit(self.generate_batch, prompt, size): i
                for i, (prompt, size) in enumerate(tasks)
            }

            # 收集结果
            with tqdm(total=len(futures), desc="生成进度") as pbar:
                for future in as_completed(futures):
                    try:
                        batch_data = future.result()

                        # 线程安全地更新dataset
                        with self.lock:
                            dataset.extend(batch_data)

                        pbar.update(1)
                        pbar.set_postfix({
                            'generated': len(dataset),
                            'cost': f"${self.stats['total_cost']:.2f}"
                        })

                    except Exception as e:
                        self.logger.error(f"批次失败: {e}")

        self.stats['end_time'] = time.time()
        self._print_summary()

        if save_path:
            self._save_final(dataset, save_path)

        return dataset
```

```python
# src/filter.py
import re
from typing import List, Dict, Callable

class DataFilter:
    """数据质量过滤器"""

    def __init__(self, config: Dict):
        self.config = config

    def filter_dataset(self, dataset: List[Dict]) -> List[Dict]:
        """
        完整过滤管道

        返回: 过滤后的数据集
        """
        print(f"原始数据: {len(dataset)}条")

        # 第1层: 格式检查
        dataset = self.format_check(dataset)
        print(f"格式检查后: {len(dataset)}条")

        # 第2层: 内容检查
        dataset = self.content_check(dataset)
        print(f"内容检查后: {len(dataset)}条")

        # 第3层: 去重
        dataset = self.deduplication(dataset)
        print(f"去重后: {len(dataset)}条")

        # 第4层: 质量打分(可选,较慢)
        if self.config['filtering']['llm_quality_check']:
            dataset = self.quality_scoring(dataset)
            print(f"质量打分后: {len(dataset)}条")

        return dataset

    def format_check(self, dataset: List[Dict]) -> List[Dict]:
        """格式检查"""
        required_fields = self.config['filtering']['required_fields']
        valid_data = []

        for item in dataset:
            # 检查必需字段存在
            if all(field in item for field in required_fields):
                # 检查字段非空
                if all(item[field] and len(str(item[field]).strip()) > 0 for field in required_fields):
                    valid_data.append(item)

        return valid_data

    def content_check(self, dataset: List[Dict]) -> List[Dict]:
        """内容检查"""
        min_len = self.config['filtering']['min_length']
        max_len = self.config['filtering']['max_length']
        max_repeat = self.config['filtering']['max_repeat_chars']

        filtered = []

        for item in dataset:
            # 获取主要文本字段
            text = self._get_main_text(item)

            # 长度检查
            if not (min_len <= len(text) <= max_len):
                continue

            # 重复字符检查
            if self._has_repeated_chars(text, max_repeat):
                continue

            # 特殊字符比例检查
            if self._special_char_ratio(text) > 0.3:
                continue

            # 敏感词检查
            if self._contains_sensitive_words(text):
                continue

            filtered.append(item)

        return filtered

    def _get_main_text(self, item: Dict) -> str:
        """获取数据项的主要文本内容"""
        # 根据数据类型返回主要文本字段
        if 'text' in item:
            return item['text']
        elif 'question' in item:
            return item['question']
        elif 'dialogue' in item:
            # 对话类型,拼接所有轮次
            return ' '.join([turn['content'] for turn in item['dialogue']])
        return ''

    def _has_repeated_chars(self, text: str, max_repeat: int) -> bool:
        """检查是否有过多重复字符"""
        pattern = r'(.)\1{' + str(max_repeat) + r',}'
        return bool(re.search(pattern, text))

    def _special_char_ratio(self, text: str) -> float:
        """计算特殊字符比例"""
        special_chars = sum(1 for c in text if not c.isalnum() and not c.isspace())
        return special_chars / len(text) if text else 0

    def _contains_sensitive_words(self, text: str) -> bool:
        """敏感词检查"""
        # 实际应用中应加载完整敏感词库
        sensitive_words = ['敏感词1', '敏感词2']
        return any(word in text for word in sensitive_words)

    def deduplication(self, dataset: List[Dict]) -> List[Dict]:
        """去重"""
        seen = set()
        unique_data = []

        for item in dataset:
            # 使用主要文本作为去重key
            text = self._get_main_text(item).strip().lower()

            # 计算文本hash
            text_hash = hash(text)

            if text_hash not in seen:
                seen.add(text_hash)
                unique_data.append(item)

        return unique_data

    def quality_scoring(self, dataset: List[Dict]) -> List[Dict]:
        """
        LLM质量打分

        注意: 这个操作会调用LLM,成本较高!
        """
        threshold = self.config['filtering']['quality_threshold']
        high_quality = []

        # 为了节省成本,可以只抽样评估部分数据
        sample_rate = 0.2  # 评估20%数据
        import random
        sample_size = int(len(dataset) * sample_rate)
        sample_indices = set(random.sample(range(len(dataset)), sample_size))

        for i, item in enumerate(tqdm(dataset, desc="质量评估")):
            # 只评估抽样数据
            if i not in sample_indices:
                high_quality.append(item)  # 未评估的默认保留
                continue

            score = self._llm_quality_score(item)

            if score >= threshold:
                item['quality_score'] = score
                high_quality.append(item)

        return high_quality

    def _llm_quality_score(self, item: Dict) -> float:
        """使用LLM评分"""
        import openai

        text = self._get_main_text(item)
```

```python
def estimate_cost(
    num_samples: int,
    model: str = 'gpt-4',
    avg_input_tokens: int = 500,
    avg_output_tokens: int = 300
) -> float:
    """
    预估生成成本

    参数:
        num_samples: 样本数量
        model: 模型名称
        avg_input_tokens: 平均输入tokens
        avg_output_tokens: 平均输出tokens

    返回:
        预估成本(美元)
    """
    if model == 'gpt-4':
        input_cost_per_1k = 0.03
        output_cost_per_1k = 0.06
    else:  # gpt-3.5-turbo
        input_cost_per_1k = 0.001
        output_cost_per_1k = 0.002

    total_input_tokens = num_samples * avg_input_tokens
    total_output_tokens = num_samples * avg_output_tokens

    input_cost = (total_input_tokens / 1000) * input_cost_per_1k
    output_cost = (total_output_tokens / 1000) * output_cost_per_1k

    total_cost = input_cost + output_cost

    return total_cost
```

```python
            (self.metrics['avg_latency'] * (self.metrics['api_calls'] - 1) + latency)
            / self.metrics['api_calls']
        )

    def calculate_throughput(self, total_samples: int, total_time: float):
```
