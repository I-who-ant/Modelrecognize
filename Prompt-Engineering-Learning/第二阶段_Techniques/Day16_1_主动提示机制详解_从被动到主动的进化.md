## 核心概念：一句话解释

**主动提示（Active Prompt）**就是：
```
让AI主动发现提示中的问题 → 主动提问澄清 → 主动生成示例 → 主动优化提示
```

**本质**：从"被动接受指令"到"主动完善理解"！

### Stage 4: 提示迭代优化 🔄
```python
class PromptOptimizer:
    def iterative_optimize(self, prompt, clarifications, examples):
        """多轮迭代优化"""
        current_prompt = prompt
        optimization_log = []

        for round in range(3):  # 最多3轮
            # 评估当前提示
            score = self.evaluate_prompt(current_prompt)

            # 基于澄清和示例优化
            improved_prompt = self.llm.generate(f"""
            优化以下提示：
            当前提示：{current_prompt}
            澄清信息：{clarifications}
            示例：{examples}

            优化要求：
            1. 整合所有澄清信息
            2. 合理安排示例
            3. 消除模糊表述
            4. 保持清晰结构

            优化后提示：
            """)

            # 检查改进
            new_score = self.evaluate_prompt(improved_prompt)
            if new_score > score:
                current_prompt = improved_prompt
                optimization_log.append(f"第{round+1}轮优化：+{new_score-score:.2f}")

        return current_prompt, optimization_log
```

## 核心代码实现

### 基础主动提示系统
```python
class BasicActivePrompt:
    """基础主动提示系统"""
    def __init__(self, llm):
        self.llm = llm
        self.uncertainty_detector = UncertaintyDetector(llm)
        self.question_generator = QuestionGenerator(llm)
        self.example_generator = ExampleGenerator(llm)
        self.prompt_optimizer = PromptOptimizer(llm)

    def主动_optimize(self, initial_prompt, task, max_rounds=3):
        """主动优化提示"""
        print("🚀 开始主动提示优化...")

        current_prompt = initial_prompt
        optimization_history = []

        for round_num in range(max_rounds):
            print(f"\n=== 第 {round_num + 1} 轮优化 ===")

            # 1. 检测不确定性
            print("🔍 检测不确定性...")
            uncertainties = self.uncertainty_detector.detect(current_prompt, task)
            print(f"   发现 {len(uncertainties)} 个问题")

            if not uncertainties:
                print("   ✅ 未发现明显问题，优化完成")
                break

            # 2. 生成澄清问题
            print("❓ 生成澄清问题...")
            questions = self.question_generator.generate(uncertainties)
            for i, q in enumerate(questions, 1):
                print(f"   问题{i}: {q}")

            # 3. 收集澄清信息
            print("💡 收集澄清信息...")
            clarifications = self.collect_clarifications(questions)

            # 4. 生成补充示例
            print("📚 生成补充示例...")
            examples = self.example_generator.generate(task, uncertainties)

            # 5. 优化提示
            print("🔧 优化提示...")
            improved_prompt = self.prompt_optimizer.optimize(
                current_prompt, clarifications, examples
            )

            # 6. 评估改进效果
            improvement = self.evaluate_improvement(
                current_prompt, improved_prompt, task
            )

            print(f"   改进幅度: {improvement:.4f}")

            # 记录优化历史
            optimization_history.append({
                "round": round_num + 1,
                "uncertainties": uncertainties,
                "questions": questions,
                "clarifications": clarifications,
                "examples": examples,
                "improvement": improvement,
                "optimized_prompt": improved_prompt
            })

            # 如果有显著改进，更新当前提示
            if improvement > 0.05:  # 改进超过5%
                current_prompt = improved_prompt
                print("   ✅ 接受优化")
            else:
                print("   ❌ 改进不明显，停止优化")
                break

        return {
            "optimized_prompt": current_prompt,
            "optimization_history": optimization_history,
            "total_improvement": (
                optimization_history[-1]["improvement"]
                if optimization_history else 0
            )
        }

    def collect_clarifications(self, questions):
        """收集澄清信息（模拟实现）"""
        clarifications = []
        for question in questions:
            # 实际应用中这里会与用户交互
            clarification = self.llm.generate(f"""
            基于以下问题，提供澄清信息：

            问题：{question}

            请提供：
            1. 直接回答
            2. 补充细节
            3. 注意事项

            澄清：
            """, max_tokens=200)
            clarifications.append(clarification)
        return clarifications

    def evaluate_improvement(self, old_prompt, new_prompt, task):
        """评估改进效果"""
        # 简化的评估逻辑
        old_score = self.calculate_prompt_score(old_prompt, task)
        new_score = self.calculate_prompt_score(new_prompt, task)
        return new_score - old_score

    def calculate_prompt_score(self, prompt, task):
        """计算提示评分"""
        score_factors = {
            "clarity": self.assess_clarity(prompt),  # 清晰度
            "completeness": self.assess_completeness(prompt, task),  # 完整性
            "actionability": self.assess_actionability(prompt)  # 可执行性
        }

        weights = {"clarity": 0.3, "completeness": 0.4, "actionability": 0.3}
        return sum(score_factors[factor] * weights[factor] for factor in score_factors)

    def assess_clarity(self, prompt):
        """评估清晰度"""
        clarity_indicators = ["明确", "具体", "清晰", "详细"]
        vague_indicators = ["可能", "也许", "大概", "适当"]

        positive = sum(1 for indicator in clarity_indicators if indicator in prompt)
        negative = sum(1 for indicator in vague_indicators if indicator in prompt)

        return max(0, min((positive - negative) / 5, 1.0))

    def assess_completeness(self, prompt, task):
        """评估完整性"""
        required_elements = ["任务", "输入", "输出", "要求", "示例"]
        return min(sum(1 for element in required_elements if element in prompt) / len(required_elements), 1.0)

    def assess_actionability(self, prompt):
        """评估可执行性"""
        action_indicators = ["步骤", "流程", "方法", "操作"]
        return min(sum(1 for indicator in action_indicators if indicator in prompt) / 4, 1.0)
```

### APE vs 主动提示

| 维度 | APE (自动提示工程师) | 主动提示 (Active Prompt) |
|------|----------------------|-------------------------|
| **优化方式** | 搜索/进化/强化学习 | 交互对话 |
| **信息获取** | 算法生成 | 主动提问 |
| **交互对象** | 模型内部 | 用户/开发者 |
| **优化速度** | 快（自动化） | 中等（需要交互） |
| **理解深度** | 基于数据 | 基于对话 |
| **适用场景** | 标准化任务 | 复杂理解任务 |

### ART vs 主动提示

| 维度 | ART (自动推理工具) | 主动提示 (Active Prompt) |
|------|-------------------|-------------------------|
| **核心功能** | 任务分解和工具使用 | 提示优化和澄清 |
| **工作对象** | 复杂任务执行 | 指令质量改进 |
| **交互方式** | 工具调用链 | 问答对话 |
| **优化目标** | 任务完成效果 | 提示清晰度 |
