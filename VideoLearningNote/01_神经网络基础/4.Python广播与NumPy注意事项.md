**NumPy向量注意事项**: 一维数组 `(n,)` 与二维数组 `(n,1)` 或 `(1,n)` 的行为差异,是常见bug的来源。

**核心目标**:

### 1. 什么是广播?

**定义**: 当两个数组形状不同时,NumPy自动扩展较小的数组,使其形状兼容。

**目的**: 避免显式循环和内存复制,提高效率。

### 2. 广播的基本示例

#### 示例1: 标量与数组

```python
import numpy as np

# 数组 + 标量
A = np.array([[1, 2, 3],
              [4, 5, 6]])  # (2, 3)
b = 10  # 标量

result = A + b

print("原理:")
print("A + b")
print("(2, 3) + 标量")

print("[[1, 2, 3],    [[10, 10, 10],    [[11, 12, 13],")
print(" [4, 5, 6]]  +  [10, 10, 10]]  =  [14, 15, 16]]")

print(f"实际结果:\n{result}")
```

### 3. 广播规则 ⭐⭐⭐ (核心!)

**NumPy广播的三条规则**:

```
规则1: 维度对齐
  从最右边的维度开始比较
  如果维度数不同,在较少维度的数组前面补1

规则2: 维度兼容性
  两个数组在某个维度兼容,当且仅当:
  - 它们在该维度上的大小相同
  - 或者其中一个的大小为1

规则3: 扩展维度
  大小为1的维度会被扩展(复制)到另一个数组的大小
```

#### 规则详细解释

```python
import numpy as np

# 例1: (2, 3) + (3,)
A = np.array([[1, 2, 3],
              [4, 5, 6]])  # (2, 3)
v = np.array([10, 20, 30])  # (3,)

# 步骤1: 对齐维度(从右向左)
#   A: (2, 3)
#   v:    (3,)  ← 前面补1 → (1, 3)

# 步骤2: 检查兼容性
#   维度0: 2 vs 1 → 兼容 (1可以扩展到2)
#   维度1: 3 vs 3 → 兼容 (相同)

# 步骤3: 扩展
#   v: (1, 3) → (2, 3)
#   [[10, 20, 30]] → [[10, 20, 30],
#                     [10, 20, 30]]

result = A + v
print("例1: (2,3) + (3,)")
print(f"结果形状: {result.shape}")  # (2, 3)
print()

# 例2: (2, 3) + (2, 1)
B = np.array([[1, 2, 3],
              [4, 5, 6]])  # (2, 3)
w = np.array([[10],
              [20]])  # (2, 1)

# 步骤1: 对齐维度
#   B: (2, 3)
#   w: (2, 1)

# 步骤2: 检查兼容性
#   维度0: 2 vs 2 → 兼容 (相同)
#   维度1: 3 vs 1 → 兼容 (1可以扩展到3)

# 步骤3: 扩展
#   w: (2, 1) → (2, 3)
#   [[10],    → [[10, 10, 10],
#    [20]]       [20, 20, 20]]

result = B + w
print("例2: (2,3) + (2,1)")
print(f"结果形状: {result.shape}")  # (2, 3)
```

### 5. 广播失败的情况 ❌

```python
import numpy as np

print("=" * 60)
print("广播失败的例子")
print("=" * 60)

# 例1: 维度不兼容
try:
    A = np.array([[1, 2, 3],
                  [4, 5, 6]])  # (2, 3)
    B = np.array([[1, 2],
                  [3, 4]])  # (2, 2)
    result = A + B
except ValueError as e:
    print(f"\n例1失败: (2,3) + (2,2)")
    print(f"错误: {e}")
    print("原因: 维度1不兼容 (3 vs 2, 都不为1)")

# 例2: 完全不兼容
try:
    A = np.array([[1, 2, 3]])  # (1, 3)
    B = np.array([[1], [2], [3], [4]])  # (4, 1)
    result = A + B  # 这个能成功! (1,3) + (4,1) → (4,3)
    print(f"\n例2成功: (1,3) + (4,1) → {result.shape}")
    print("广播规则:")
    print("  维度0: 1 vs 4 → 1扩展到4")
    print("  维度1: 3 vs 1 → 1扩展到3")
except ValueError as e:
    print(f"失败: {e}")

# 例3: 真正失败的例子
try:
    A = np.array([[1, 2, 3],
                  [4, 5, 6]])  # (2, 3)
    B = np.array([[[1]], [[2]], [[3]], [[4]]])  # (4, 1, 1)
    result = A + B
except ValueError as e:
    print(f"\n例3失败: (2,3) + (4,1,1)")
    print(f"错误信息: operands could not be broadcast together")
    print("原因: 从右向左对齐:")
    print("  A:     (2, 3)")
    print("  B: (4, 1, 1)")
    print("  维度0: 2 vs 4 → 不兼容!")
```

### 6. 广播的实际应用

#### 应用1: 归一化

```python
import numpy as np

# 数据: 每列是一个特征
X = np.array([[1, 2, 100],
              [3, 4, 200],
              [5, 6, 300]])  # (3, 3)

print("原始数据:")
print(X)
print()

# 计算每列的均值
mean = X.mean(axis=0)  # (3,) 每列的均值
print(f"均值: {mean}")  # [3. 4. 200.]

# 计算每列的标准差
std = X.std(axis=0)  # (3,)
print(f"标准差: {std}")  # [1.63299316 1.63299316 81.64965809]
print()

# 标准化 (利用广播!)
X_normalized = (X - mean) / std
#              (3,3) - (3,) → (3,3) - (1,3) → 广播
#              (3,3) / (3,) → (3,3) / (1,3) → 广播

print("标准化后:")
print(X_normalized)
print()
print("验证: 每列均值≈0, 标准差≈1")
print(f"均值: {X_normalized.mean(axis=0)}")
print(f"标准差: {X_normalized.std(axis=0)}")
```

#### 应用2: Softmax (逐行)

```python
import numpy as np

def softmax(X):
    """
    对每一行计算softmax
    X: (m, n) m个样本, n个特征
    """
    # 数值稳定技巧: 减去每行最大值
    X_max = X.max(axis=1, keepdims=True)  # (m, 1)
    exp_X = np.exp(X - X_max)  # 广播: (m,n) - (m,1)
    sum_exp = exp_X.sum(axis=1, keepdims=True)  # (m, 1)
    return exp_X / sum_exp  # 广播: (m,n) / (m,1)

# 3个样本, 4个类别
logits = np.array([[2.0, 1.0, 0.1, 3.0],
                   [1.0, 2.0, 3.0, 0.1],
                   [0.5, 0.5, 0.5, 0.5]])  # (3, 4)

probs = softmax(logits)

print("Logits (得分):")
print(logits)
print()
print("Softmax概率:")
print(probs)
print()
print("每行概率和:")
print(probs.sum(axis=1))  # 应该都是1
```

**输出**:
```
Logits (得分):
[[2.  1.  0.1 3. ]
 [1.  2.  3.  0.1]
 [0.5 0.5 0.5 0.5]]

Softmax概率:
[[0.26762315 0.09844716 0.03995023 0.59397946]
 [0.09003057 0.24472847 0.66524096 0.        ]
 [0.25       0.25       0.25       0.25      ]]

每行概率和:
[1. 1. 1.]
```

#### 应用3: 逻辑回归中的广播

```python
import numpy as np

# 逻辑回归前向传播
n = 3  # 特征数
m = 5  # 样本数
 
X = np.random.randn(n, m)  # (3, 5)
w = np.random.randn(n, 1)  # (3, 1)
b = 0.5  # 标量

# 计算 Z = w^T X + b
Z = w.T @ X + b  # (1,3)@(3,5) + 标量 → (1,5) + 标量 → (1,5)
#                     ↑ 广播!

print(f"X.shape: {X.shape}")  # (3, 5)
print(f"w.shape: {w.shape}")  # (3, 1)
print(f"w.T.shape: {w.T.shape}")  # (1, 3)
print(f"Z.shape: {Z.shape}")  # (1, 5)
print()
print("广播发生在:")
print("  w.T @ X 得到 (1, 5)")
print("  (1, 5) + 标量 → 标量广播到 (1, 5)")
```

### 1. 一维数组的陷阱 ❌

**最大的坑**: 一维数组 `(n,)` 的转置不起作用!

```python

print("=" * 60)
print("一维数组的陷阱")
```

### 5. 使用断言检查维度 ⭐⭐⭐

**最佳实践**: 在关键位置添加断言检查!

```python
import numpy as np

def safe_forward_propagation(X, w, b):
    """
    安全的前向传播,带维度检查

    X: (n, m) 特征矩阵
    w: (n, 1) 权重列向量
    b: 标量
    """
    # 断言检查
    assert len(X.shape) == 2, f"X应该是2维,实际是{len(X.shape)}维"
    assert len(w.shape) == 2, f"w应该是2维,实际是{len(w.shape)}维"
    assert w.shape[1] == 1, f"w应该是列向量(n,1),实际是{w.shape}"

    n, m = X.shape
    assert w.shape[0] == n, f"w维度应为({n},1),实际是{w.shape}"

    # 计算
    Z = w.T @ X + b  # (1, n) @ (n, m) + 标量 = (1, m)

    # 后置检查
    assert Z.shape == (1, m), f"Z形状应为(1,{m}),实际是{Z.shape}"

    return Z

# 测试
X = np.random.randn(3, 5)
w = np.random.randn(3, 1)
b = 0.5

Z = safe_forward_propagation(X, w, b)
print(f"✅ 前向传播成功! Z.shape: {Z.shape}")

# 测试错误情况
try:
    w_wrong = np.random.randn(3)  # 一维数组!
    Z = safe_forward_propagation(X, w_wrong, b)
except AssertionError as e:
    print(f"\n❌ 捕获错误: {e}")
```

## 综合示例: 正确的逻辑回归实现

```python
import numpy as np

def sigmoid(z):
    """Sigmoid激活函数"""
    return 1 / (1 + np.exp(-z))

def initialize_parameters(n_features):
    """
    初始化参数 (正确版本!)

    参数:
    n_features: 特征数

    返回:
    w: (n_features, 1) 列向量
    b: 标量
    """
    w = np.zeros((n_features, 1))  # ✅ 二维列向量
    b = 0.0

    # 断言检查
    assert w.shape == (n_features, 1), f"w形状错误: {w.shape}"

    return w, b

def forward_propagation(X, w, b):
    """
    前向传播 (带完整检查!)

    参数:
    X: (n_features, m_samples)
    w: (n_features, 1)
    b: 标量

    返回:
    A: (1, m_samples) 预测概率
    """
    # 前置检查
    assert len(X.shape) == 2, "X必须是2维"
    assert len(w.shape) == 2, "w必须是2维"
    assert w.shape[1] == 1, "w必须是列向量"

    n, m = X.shape
    assert w.shape[0] == n, f"w维度不匹配: {w.shape} vs X: {X.shape}"

    # 计算
    Z = w.T @ X + b  # (1, n) @ (n, m) + 标量 = (1, m)
    A = sigmoid(Z)    # (1, m)

    # 后置检查
    assert Z.shape == (1, m), f"Z形状错误: {Z.shape}"
    assert A.shape == (1, m), f"A形状错误: {A.shape}"

    return A

def compute_gradients(X, A, Y):
    """
    计算梯度 (向量化 + 广播!)

    参数:
    X: (n_features, m_samples)
    A: (1, m_samples) 预测
    Y: (1, m_samples) 真实标签

    返回:
    dw: (n_features, 1)
    db: 标量
    """
    n, m = X.shape

    # 检查
    assert A.shape == (1, m), f"A形状错误: {A.shape}"
    assert Y.shape == (1, m), f"Y形状错误: {Y.shape}"

    # 计算梯度
    dZ = A - Y              # (1, m) 广播相减
    dw = 1/m * X @ dZ.T     # (n, m) @ (m, 1) = (n, 1) ✅
    db = 1/m * np.sum(dZ)   # 标量

    # 检查
    assert dw.shape == (n, 1), f"dw形状错误: {dw.shape}"

    return dw, db

# 完整示例
print("=" * 60)
print("完整的逻辑回归示例")
print("=" * 60)

# 生成数据
n_features = 10
m_samples = 100

X = np.random.randn(n_features, m_samples)
Y = (np.random.rand(1, m_samples) > 0.5).astype(float)  # ✅ (1, m)

# 初始化
w, b = initialize_parameters(n_features)
print(f"✅ 初始化: w.shape={w.shape}, b={b}")

# 前向传播
A = forward_propagation(X, w, b)
print(f"✅ 前向传播: A.shape={A.shape}")

# 计算梯度
dw, db = compute_gradients(X, A, Y)
print(f"✅ 计算梯度: dw.shape={dw.shape}, db={db:.6f}")

# 更新参数
alpha = 0.01
w = w - alpha * dw
b = b - alpha * db
print(f"✅ 参数更新完成")

print("\n所有维度检查通过! ✅")
```

## 总结

### 广播规则记忆

```python
【广播三步法】
1. 对齐维度(从右向左,不足补1)
2. 检查兼容(相同或其中之一为1)
3. 扩展维度(1扩展到对应大小)

【常见模式】
(m, n) + (n,)   → (m, n) ✅  行向量广播
(m, n) + (m, 1) → (m, n) ✅  列向量广播
(m, n) + 标量   → (m, n) ✅  标量广播
(1, n) + (m, 1) → (m, n) ✅  外积广播
```

### NumPy向量最佳实践

```python
【黄金规则】
1. 永远不要使用一维数组 (n,) 进行矩阵运算
2. 总是使用二维数组: (n, 1) 或 (1, n)
3. 添加 assert 检查维度
4. 使用 reshape 明确转换

【检查清单】
✅ 所有权重都是 (n, 1) 或 (1, n)
✅ 所有数据都是 (n, m) 形式
✅ 关键位置有 assert 检查
✅ 避免隐式转换和一维数组
```

### 常见错误对照表

| 错误 ❌ | 正确 ✅ | 说明 |
|--------|---------|------|
| `w = np.random.randn(n)` | `w = np.random.randn(n, 1)` | 使用列向量 |
| `a.T` 当 a.shape=(n,) | `a.reshape(1, -1)` 或创建时就用2维 | 一维数组转置无效 |
| 不检查维度 | `assert w.shape == (n, 1)` | 添加断言 |
| `X.shape = (m, n)` | `X.shape = (n, m)` | 特征×样本 |
