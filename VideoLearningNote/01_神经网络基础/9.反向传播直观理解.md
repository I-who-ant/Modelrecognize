## 基本定义

**反向传播(Backpropagation)**: 利用链式法则,从输出层开始向输入层逐层计算损失函数对每个参数的梯度的算法。

**核心思想**:
- 前向传播: 输入 → 输出 → 损失
- 反向传播: 损失 → 梯度 → 参数更新
- 链式法则: 复合函数求导的规则

    反向传播核心：利用链式法则算偏导
    ```
      ∂L/∂w_ij = (∂L/∂a_j) · (∂a_j/∂z_j) · (∂z_j/∂w_ij)
  
   •  ∂L/∂a_j 是从上一层（更靠近输出）传回来的“误差信号”；
   •  ∂a_j/∂z_j 取决于激活函数；
   •  ∂z_j/∂w_ij = x_i，就是该权重对应的输入。
    ```
  
   •  每个 mini-batch 先 前向 推到输出，算出损失 L；不是直接“学习率 × 损失”，
      而是先用链式法则算出每个权重的梯度 ∂L/∂w。

   •  再用优化器做 参数更新：w ← w − η · ∂L/∂w。学习率 η 只是缩放梯度，真正决定方向的是
      ∂L/∂w。
   •  这个更新是“后面的层先得到梯度，再一层层往前传”，所以所有层都会被 更新到。
   •  更新后就用下一批训练数据继续前向/反向

### 2. 链式法则的数学原理

```python
# ============================================================
# 链式法则 (Chain Rule)
# ============================================================

"""
单变量链式法则:
如果 y = f(u), u = g(x)
那么 dy/dx = dy/du · du/dx

多变量链式法则:
如果 z = f(x, y), x = g(t), y = h(t)
那么 dz/dt = ∂z/∂x · dx/dt + ∂z/∂y · dy/dt

神经网络中的链式法则:
L = loss(a[L])
a[L] = g(z[L])
z[L] = W[L] @ a[L-1] + b[L]

∂L/∂W[L] = ∂L/∂a[L] · ∂a[L]/∂z[L] · ∂z[L]/∂W[L]
           ↑           ↑              ↑
         从后传来    激活函数导数    本地梯度
"""

def chain_rule_visualization():
    """
    链式法则可视化
    """
    print("=" * 60)
    print("链式法则示例")
    print("=" * 60)

    # 示例: y = sin(x^2)
    # dy/dx = dy/du · du/dx = cos(u) · 2x = cos(x^2) · 2x

    x = 1.5

    # 前向传播
    u = x**2
    y = np.sin(u)

    print(f"x = {x}")
    print(f"u = x² = {u}")
    print(f"y = sin(u) = {y:.6f}")
    print()

    # 反向传播
    dy_du = np.cos(u)  # ∂y/∂u = cos(u)
    du_dx = 2 * x      # ∂u/∂x = 2x
    dy_dx = dy_du * du_dx  # 链式法则

    print("反向传播:")
    print(f"∂y/∂u = cos(u) = {dy_du:.6f}")
    print(f"∂u/∂x = 2x = {du_dx:.6f}")
    print(f"∂y/∂x = ∂y/∂u · ∂u/∂x = {dy_du:.6f} · {du_dx:.6f} = {dy_dx:.6f}")
    print()

    # 验证
    epsilon = 1e-7
    dy_dx_numerical = (np.sin((x + epsilon)**2) - np.sin((x - epsilon)**2)) / (2 * epsilon)
    print(f"验证 (数值): ∂y/∂x = {dy_dx_numerical:.6f}")
    print(f"误差: {abs(dy_dx - dy_dx_numerical):.2e}")
    print()

chain_rule_visualization()
```

### 1. 单样本逻辑回归

```python
# ============================================================
# 逻辑回归反向传播 (单样本)
# ============================================================

def logistic_regression_backward_single():
    """
    逻辑回归反向传播 (单样本)

    前向传播:
    z = w^T x + b
    a = σ(z)
    L = -(y log(a) + (1-y) log(1-a))

    反向传播:
    ∂L/∂a = -y/a + (1-y)/(1-a)
    ∂L/∂z = ∂L/∂a · ∂a/∂z = (a - y)  ← 奇迹般的简化!
    ∂L/∂w = ∂L/∂z · ∂z/∂w = (a - y) · x
    ∂L/∂b = ∂L/∂z · ∂z/∂b = (a - y)
    """

    print("=" * 60)
    print("逻辑回归反向传播 (单样本)")
    print("=" * 60)

    # 输入
    x = np.array([[1.0], [2.0]])  # (2, 1)
    y = 1.0
    w = np.array([[0.5], [0.3]])  # (2, 1)
    b = 0.1

    print(f"输入: x = {x.flatten()}")
    print(f"标签: y = {y}")
    print(f"参数: w = {w.flatten()}, b = {b}")
    print()

    # ========== 前向传播 ==========
    z = np.dot(w.T, x) + b  # (1, 1)
    a = 1 / (1 + np.exp(-z))  # sigmoid
    L = -(y * np.log(a) + (1 - y) * np.log(1 - a))

    print("前向传播:")
    print(f"z = w^T x + b = {float(z):.6f}")
    print(f"a = σ(z) = {float(a):.6f}")
    print(f"L = {float(L):.6f}")
    print()

    # ========== 反向传播 (详细步骤) ==========
    print("反向传播 (详细步骤):")

    # 步骤1: ∂L/∂a
    dL_da = -(y / a) + (1 - y) / (1 - a)
    print(f"步骤1: ∂L/∂a = -{y}/a + {1-y}/(1-a) = {float(dL_da):.6f}")

    # 步骤2: ∂a/∂z (sigmoid导数)
    da_dz = a * (1 - a)
    print(f"步骤2: ∂a/∂z = a(1-a) = {float(da_dz):.6f}")

    # 步骤3: 链式法则 ∂L/∂z
    dL_dz_detailed = dL_da * da_dz
    print(f"步骤3: ∂L/∂z = ∂L/∂a · ∂a/∂z = {float(dL_da):.6f} · {float(da_dz):.6f} = {float(dL_dz_detailed):.6f}")

    # 简化形式
    dL_dz = a - y
    print(f"简化: ∂L/∂z = a - y = {float(a):.6f} - {y} = {float(dL_dz):.6f}")
    print(f"验证相同: {np.allclose(dL_dz_detailed, dL_dz)}")
    print()

    # 步骤4: ∂L/∂w
    dL_dw = dL_dz * x  # (2, 1)
    print(f"步骤4: ∂L/∂w = ∂L/∂z · x = {float(dL_dz):.6f} · {x.flatten()}")
    print(f"      = {dL_dw.flatten()}")
    print()

    # 步骤5: ∂L/∂b
    dL_db = dL_dz
    print(f"步骤5: ∂L/∂b = ∂L/∂z = {float(dL_db):.6f}")
    print()

    # ========== 参数更新 ==========
    learning_rate = 0.1
    w_new = w - learning_rate * dL_dw
    b_new = b - learning_rate * dL_db

    print(f"参数更新 (lr={learning_rate}):")
    print(f"w: {w.flatten()} → {w_new.flatten()}")
    print(f"b: {b} → {float(b_new):.6f}")
    print()

    # 验证: 损失下降
    z_new = np.dot(w_new.T, x) + b_new
    a_new = 1 / (1 + np.exp(-z_new))
    L_new = -(y * np.log(a_new) + (1 - y) * np.log(1 - a_new))

    print(f"损失变化: {float(L):.6f} → {float(L_new):.6f} (下降 {float(L - L_new):.6f})")

logistic_regression_backward_single()
```

### 2. 多样本向量化反向传播

```python
# ============================================================
# 逻辑回归反向传播 (多样本向量化)
# ============================================================

def logistic_regression_backward_vectorized():
    """
    逻辑回归反向传播 (向量化)

    前向传播:
    Z = W^T X + b    # (1, m)
    A = σ(Z)         # (1, m)
    L = -(Y log(A) + (1-Y) log(1-A))  # (1, m)
    J = (1/m) Σ L    # 平均损失

    反向传播:
    dZ = A - Y       # (1, m)
    dW = (1/m) X @ dZ^T  # (n, 1)
    db = (1/m) Σ dZ  # 标量
    """

    print("=" * 60)
    print("逻辑回归反向传播 (向量化)")
    print("=" * 60)

    # 生成数据
    np.random.seed(1)
    n = 2  # 特征数
    m = 5  # 样本数

    X = np.random.randn(n, m)
    Y = (np.random.rand(1, m) > 0.5).astype(float)
    w = np.random.randn(n, 1) * 0.01
    b = 0.0

    print(f"数据形状: X={X.shape}, Y={Y.shape}")
    print(f"参数形状: w={w.shape}, b是标量")
    print()

    # ========== 前向传播 ==========
    Z = np.dot(w.T, X) + b  # (1, m)
    A = 1 / (1 + np.exp(-Z))  # (1, m)
    cost = -np.mean(Y * np.log(A) + (1 - Y) * np.log(1 - A))

    print("前向传播:")
    print(f"Z: {Z.shape}, 前3个样本: {Z[0, :3]}")
    print(f"A: {A.shape}, 前3个样本: {A[0, :3]}")
    print(f"损失: {cost:.6f}")
    print()

    # ========== 反向传播 (向量化) ==========
    print("反向传播 (向量化):")

    # dZ = A - Y  (每个样本的梯度)
    dZ = A - Y  # (1, m)
    print(f"dZ = A - Y: {dZ.shape}")
    print(f"dZ 前3个样本: {dZ[0, :3]}")
    print()

    # dW = (1/m) X @ dZ^T
    dW = (1/m) * np.dot(X, dZ.T)  # (n, 1)
    print(f"dW = (1/m) X @ dZ^T: {dW.shape}")
    print(f"dW = {dW.flatten()}")
    print()

    # db = (1/m) Σ dZ
    db = (1/m) * np.sum(dZ)
    print(f"db = (1/m) sum(dZ) = {db:.6f}")
    print()

    # ========== 参数更新 ==========
    learning_rate = 0.5
    w -= learning_rate * dW
    b -= learning_rate * db

    # 重新计算损失
    Z_new = np.dot(w.T, X) + b
    A_new = 1 / (1 + np.exp(-Z_new))
    cost_new = -np.mean(Y * np.log(A_new) + (1 - Y) * np.log(1 - A_new))

    print(f"参数更新后:")
    print(f"损失: {cost:.6f} → {cost_new:.6f} (下降 {cost - cost_new:.6f})")

logistic_regression_backward_vectorized()
```

# ============================================================
# 2层神经网络反向传播 (单样本,详细推导)
# ============================================================

def two_layer_backward_detailed():
    """
    2层神经网络反向传播

    网络结构: x → [W1,b1,ReLU] → a1 → [W2,b2,Sigmoid] → a2

    前向传播:
    z1 = W1 @ x + b1
    a1 = ReLU(z1)
    z2 = W2 @ a1 + b2
    a2 = σ(z2)
    L = -(y log(a2) + (1-y) log(1-a2))

    反向传播 (从输出层到输入层):
    第2层 (输出层):
    dz2 = a2 - y
    dW2 = dz2 @ a1^T
    db2 = dz2
    da1 = W2^T @ dz2

    第1层 (隐藏层):
    dz1 = da1 * ReLU'(z1)  = da1 * (z1 > 0)
    dW1 = dz1 @ x^T
    db1 = dz1
    """

    print("=" * 60)
    print("2层神经网络反向传播 (单样本)")
    print("=" * 60)

    # 输入
    np.random.seed(1)
    n_x = 3  # 输入维度
    n_h = 4  # 隐藏层大小
    n_y = 1  # 输出维度

    x = np.random.randn(n_x, 1)
    y = 1.0

    # 初始化参数
    W1 = np.random.randn(n_h, n_x) * 0.01
    b1 = np.zeros((n_h, 1))
    W2 = np.random.randn(n_y, n_h) * 0.01
    b2 = np.zeros((n_y, 1))

    print(f"网络结构: {n_x} → {n_h} → {n_y}")
    print(f"输入: x={x.shape}, y={y}")
    print()

    # ========== 前向传播 ==========
    print("前向传播:")

    z1 = np.dot(W1, x) + b1
    a1 = np.maximum(0, z1)  # ReLU
    print(f"z1 = W1 @ x + b1: {z1.shape}")
    print(f"a1 = ReLU(z1): {a1.shape}")

    z2 = np.dot(W2, a1) + b2
    a2 = 1 / (1 + np.exp(-z2))  # Sigmoid
    print(f"z2 = W2 @ a1 + b2: {z2.shape}")
    print(f"a2 = σ(z2): {a2.shape}, 值={float(a2):.6f}")

    L = -(y * np.log(a2) + (1 - y) * np.log(1 - a2))
    print(f"L = {float(L):.6f}")
    print()

    # ========== 反向传播 ==========
    print("反向传播:")
    print()

    # ========== 第2层 (输出层) ==========
    print("第2层 (输出层):")

    # dL/da2
    dL_da2 = -(y / a2) + (1 - y) / (1 - a2)
    print(f"  dL/da2 = {float(dL_da2):.6f}")

    # da2/dz2 (sigmoid导数)
    da2_dz2 = a2 * (1 - a2)
    print(f"  da2/dz2 = a2(1-a2) = {float(da2_dz2):.6f}")

    # dL/dz2 (链式法则)
    dz2 = dL_da2 * da2_dz2
    print(f"  dz2 = dL/da2 · da2/dz2 = {float(dz2):.6f}")
    print(f"  简化: dz2 = a2 - y = {float(a2 - y):.6f} ✓")
    dz2 = a2 - y  # 使用简化形式
    print()

    # dL/dW2
    dW2 = np.dot(dz2, a1.T)
    print(f"  dW2 = dz2 @ a1^T: {dW2.shape}")
    print(f"  dW2 = {dW2.flatten()[:3]}... (前3个)")
    print()

    # dL/db2
    db2 = dz2
    print(f"  db2 = dz2: {db2.shape}")
    print(f"  db2 = {float(db2):.6f}")
    print()

    # dL/da1 (传递给上一层)
    da1 = np.dot(W2.T, dz2)
    print(f"  da1 = W2^T @ dz2: {da1.shape}")
    print(f"  da1 = {da1.flatten()[:3]}... (前3个)")

    # ========== 第1层 (隐藏层) ==========
    print("第1层 (隐藏层):")

    # da1/dz1 (ReLU导数)
    da1_dz1 = (z1 > 0).astype(float)
    print(f"  da1/dz1 = (z1 > 0): {da1_dz1.shape}")
    print(f"  da1/dz1 = {da1_dz1.flatten()[:3]}... (前3个)")
    print()

    # dL/dz1 (链式法则)
    dz1 = da1 * da1_dz1
    print(f"  dz1 = da1 * da1/dz1: {dz1.shape}")
    print(f"  dz1 = {dz1.flatten()[:3]}... (前3个)")
    print()

    # dL/dW1
    dW1 = np.dot(dz1, x.T)
    print(f"  dW1 = dz1 @ x^T: {dW1.shape}")
    print(f"  dW1 前3个神经元的梯度:")
    print(f"    {dW1[0, :]}")
    print()

    # dL/db1
    db1 = dz1
    print(f"  db1 = dz1: {db1.shape}")
    print(f"  db1 = {db1.flatten()[:3]}... (前3个)")
    print()

    # ========== 梯度汇总 ==========
    print("梯度汇总:")
    grads = {
        'dW1': dW1,
        'db1': db1,
        'dW2': dW2,
        'db2': db2
    }

    for key in grads:
        print(f"  {key}: {grads[key].shape}, "
              f"mean={np.mean(np.abs(grads[key])):.6f}, "
              f"max={np.max(np.abs(grads[key])):.6f}")
    print()

    # ========== 梯度检验 ==========
    print("梯度检验 (数值梯度):")

    def compute_loss(W1, b1, W2, b2, x, y):
        """计算损失"""
        z1 = np.dot(W1, x) + b1
        a1 = np.maximum(0, z1)
        z2 = np.dot(W2, a1) + b2
        a2 = 1 / (1 + np.exp(-z2))
        L = -(y * np.log(a2) + (1 - y) * np.log(1 - a2))
        return float(L)

    epsilon = 1e-7

    # 检验 dW2[0,0]
    W2_plus = W2.copy()
    W2_plus[0, 0] += epsilon
    W2_minus = W2.copy()
    W2_minus[0, 0] -= epsilon

    dW2_numerical = (compute_loss(W1, b1, W2_plus, b2, x, y) -
                     compute_loss(W1, b1, W2_minus, b2, x, y)) / (2 * epsilon)

    print(f"  dW2[0,0]: 解析={dW2[0, 0]:.8f}, 数值={dW2_numerical:.8f}, "
          f"差异={abs(dW2[0, 0] - dW2_numerical):.2e}")

    输出层 (L):
    dZ[L] = A[L] - Y

    dZ[l] = dA[l] * g'[l](Z[l])
    dW[l] = (1/m) dZ[l] @ A[l-1]^T
    db[l] = (1/m) sum(dZ[l], axis=1, keepdims=True)
    """

    print("多层神经网络向量化反向传播")
    print("=" * 60)

    # 网络结构
    layer_dims = [3, 4, 5, 1]  # 3 → 4 → 5 → 1
    m = 10  # 样本数

    print(f"网络结构: {' → '.join(map(str, layer_dims))}")
    print(f"样本数: {m}")
    print()

    # 初始化参数
    np.random.seed(1)
    parameters = {}
    L = len(layer_dims) - 1

    for l in range(1, L + 1):
        parameters[f'W{l}'] = np.random.randn(layer_dims[l], layer_dims[l-1]) * 0.01
        parameters[f'b{l}'] = np.zeros((layer_dims[l], 1))

    # 生成数据
    X = np.random.randn(layer_dims[0], m)
    Y = (np.random.rand(1, m) > 0.5).astype(float)

    # ========== 前向传播 (保存cache) ==========
    print("前向传播:")

    caches = []
    A = X

    for l in range(1, L + 1):
        A_prev = A
        W = parameters[f'W{l}']
        b = parameters[f'b{l}']

        Z = np.dot(W, A_prev) + b

        if l < L:
            A = np.maximum(0, Z)  # ReLU
        else:
            A = 1 / (1 + np.exp(-Z))  # Sigmoid

        cache = (A_prev, W, b, Z)
        caches.append(cache)

        print(f"  第{l}层: Z{l}={Z.shape}, A{l}={A.shape}")

    # 计算损失
    AL = A
    cost = -np.mean(Y * np.log(AL) + (1 - Y) * np.log(1 - AL))
    print(f"\n损失: {cost:.6f}")
    print()

    # ========== 反向传播 ==========
    print("反向传播:")

    grads = {}

    # 输出层
    print(f"  第{L}层 (输出层):")
    dZL = AL - Y
    A_prev_L, WL, bL, ZL = caches[L-1]

    grads[f'dW{L}'] = (1/m) * np.dot(dZL, A_prev_L.T)
    grads[f'db{L}'] = (1/m) * np.sum(dZL, axis=1, keepdims=True)
    dA_prev = np.dot(WL.T, dZL)

    print(f"    dZ{L}: {dZL.shape}")
    print(f"    dW{L}: {grads[f'dW{L}'].shape}")
    print(f"    db{L}: {grads[f'db{L}'].shape}")
    print()

    # 隐藏层
    for l in reversed(range(1, L)):
        print(f"  第{l}层 (隐藏层):")

        A_prev, W, b, Z = caches[l-1]

        # ReLU导数
        dZ = dA_prev * (Z > 0)

        grads[f'dW{l}'] = (1/m) * np.dot(dZ, A_prev.T)
        grads[f'db{l}'] = (1/m) * np.sum(dZ, axis=1, keepdims=True)

        if l > 1:
            dA_prev = np.dot(W.T, dZ)

        print(f"    dZ{l}: {dZ.shape}")
        print(f"    dW{l}: {grads[f'dW{l}'].shape}")
        print(f"    db{l}: {grads[f'db{l}'].shape}")
        print()

    # 梯度统计
    print("梯度统计:")
    for key in sorted(grads.keys()):
        grad = grads[key]
        print(f"  {key}: shape={grad.shape}, "
              f"mean={np.mean(np.abs(grad)):.6f}, "
              f"max={np.max(np.abs(grad)):.6f}")

### 1. 反向传播的本质

```python
intuition = """
============================================================
反向传播的直观理解
============================================================

1. 前向传播 = 计算预测
   输入 → 第1层 → 第2层 → ... → 输出 → 损失
   像工厂流水线: 原料 → 加工 → 成品 → 质检

2. 反向传播 = 分配责任
   损失 ← 第1层 ← 第2层 ← ... ← 输出
   像追责: 质量问题 ← 找原因 ← 追溯源头

3. 梯度 = 影响力
   ∂L/∂W[l] = 这个参数对最终损失的影响有多大
   影响大 → 梯度大 → 需要大幅调整
   影响小 → 梯度小 → 只需微调

4. 链式法则 = 逐层传递
   ∂L/∂W[1] = ∂L/∂A[L] · ∂A[L]/∂Z[L] · ... · ∂Z[2]/∂A[1] · ∂A[1]/∂W[1]

   像传话游戏:
   最后的人 → 倒数第二个 → ... → 第一个人
   每层都把"责任"传递给上一层

5. 为什么从后向前?
   因为损失在最后!
   - 损失告诉我们哪里错了
   - 从损失开始,逐层往回找原因
   - 每层都问: "我对损失有多少责任?"

============================================================
"""

print(intuition)
```

## 常见疑问

### Q1: 为什么叫"反向"传播?

**A**: 因为计算方向与前向相反

```python
"""
前向传播方向:
输入 → 第1层 → 第2层 → ... → 输出 → 损失
(从左到右)

反向传播方向:
输入 ← 第1层 ← 第2层 ← ... ← 输出 ← 损失
(从右到左)

为什么要反向?
- 损失在最后
- 需要从损失往回计算每层的梯度
- 链式法则决定了必须从后往前
"""
```

### Q2: 反向传播和梯度下降的关系?

**A**: 反向传播计算梯度,梯度下降更新参数

```python
"""
完整训练循环:

1. 前向传播:
   计算预测值和损失

2. 反向传播:  ← 计算梯度
   计算 ∂L/∂W, ∂L/∂b

3. 梯度下降:  ← 更新参数
   W = W - α · ∂L/∂W
   b = b - α · ∂L/∂b

关系:
- 反向传播 = 求导 (计算梯度)
- 梯度下降 = 优化 (使用梯度)
- 反向传播提供方向
- 梯度下降执行移动
"""
```

### Q4: 如何验证反向传播正确?

**A**: 梯度检验

```python
def gradient_check(parameters, grads, X, Y, epsilon=1e-7):
    """
    梯度检验: 用数值梯度验证解析梯度

    数值梯度:
    ∂L/∂W ≈ (L(W+ε) - L(W-ε)) / (2ε)

    应该与解析梯度接近
    """

    for key in parameters:
        param = parameters[key]
        grad_analytic = grads[f'd{key}']

        # 随机选几个参数检查
        for i in range(min(5, param.size)):
            # 获取位置
            idx = np.unravel_index(i, param.shape)

            # L(W + ε)
            param[idx] += epsilon
            loss_plus = compute_loss_forward(X, Y, parameters)

            # L(W - ε)
            param[idx] -= 2 * epsilon
            loss_minus = compute_loss_forward(X, Y, parameters)

            # 恢复
            param[idx] += epsilon

            # 数值梯度
            grad_numerical = (loss_plus - loss_minus) / (2 * epsilon)
            grad_analytic_val = grad_analytic[idx]

            # 相对误差
            diff = abs(grad_numerical - grad_analytic_val)
            rel_error = diff / (abs(grad_numerical) + abs(grad_analytic_val) + 1e-8)

            print(f"{key}{idx}: "
                  f"数值={grad_numerical:.8f}, "
                  f"解析={grad_analytic_val:.8f}, "
                  f"相对误差={rel_error:.2e}")

            # 检查
            if rel_error > 1e-5:
                print(f"  ⚠️  警告: 相对误差较大!")
            else:
                print(f"  ✓ OK")

# 梯度检验很慢,只在调试时用
# 训练时关闭
```

## 总结

### 核心公式

```python
# 反向传播通用公式

# 输出层 (l = L):
dZ[L] = A[L] - Y  # (交叉熵 + Sigmoid简化)
dW[L] = (1/m) dZ[L] @ A[L-1]^T
db[L] = (1/m) sum(dZ[L], axis=1, keepdims=True)

# 隐藏层 (l < L):
dA[l] = W[l+1]^T @ dZ[l+1]
dZ[l] = dA[l] * g'[l](Z[l])  # g'是激活函数导数
dW[l] = (1/m) dZ[l] @ A[l-1]^T
db[l] = (1/m) sum(dZ[l], axis=1, keepdims=True)

# 关键:
# 1. 从后往前计算
# 2. 链式法则传递梯度
# 3. 激活函数导数很重要
```

### 关键记忆点

1. **反向传播 = 链式法则**: 逐层计算梯度
2. **从输出到输入**: 损失 → ... → 参数
3. **保存前向值**: cache用于反向计算
4. **激活函数导数**: ReLU: (Z>0), Sigmoid: A(1-A)
5. **梯度 = 影响力**: 参数对损失的影响程度
6. **向量化高效**: 同时处理m个样本

### 反向传播检查清单

```python
✅ 前向传播保存cache (A_prev, W, b, Z)
✅ 从输出层开始反向
✅ 正确应用链式法则
✅ 激活函数导数正确
✅ 梯度维度匹配
✅ 除以m (平均梯度)
✅ 梯度检验通过
```
