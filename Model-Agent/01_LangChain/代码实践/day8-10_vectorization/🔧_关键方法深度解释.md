## 1️⃣ TextLoader vs DirectoryLoader - 文档加载

### TextLoader - 单文件加载

```python
from langchain_community.document_loaders import TextLoader

# 加载单个文本文件
loader = TextLoader(
    file_path="document.txt",
    encoding='utf-8'  # 重要：指定编码
)

docs = loader.load()

# 返回Document列表(通常只有1个)
print(f"加载了{len(docs)}个文档")
print(f"内容长度: {len(docs[0].page_content)}字符")
print(f"元数据: {docs[0].metadata}")
# 元数据示例: {'source': 'document.txt'}
```

### DirectoryLoader - 批量加载目录

```python
from langchain_community.document_loaders import DirectoryLoader, TextLoader

# 批量加载目录中的所有.txt文件
loader = DirectoryLoader(
    path="./documents",
    glob="*.txt",              # 文件模式(支持通配符)
    loader_cls=TextLoader,     # 使用TextLoader加载
    loader_kwargs={'encoding': 'utf-8'},
    show_progress=True         # 显示进度条
)

docs = loader.load()

print(f"加载了{len(docs)}个文档")
for doc in docs:
    print(f"  {doc.metadata['source']}: {len(doc.page_content)}字符")
```

### glob模式常见用法

```python
# 匹配所有.txt文件
glob="*.txt"

# 匹配所有.md文件
glob="*.md"

# 递归匹配所有子目录中的.txt
glob="**/*.txt"

# 匹配多种格式
glob="*"  # 所有文件

# 排除某些文件(需要结合过滤逻辑)
# DirectoryLoader不直接支持排除，可以后处理
```

### 为加载的文档添加元数据

```python
from langchain_community.document_loaders import DirectoryLoader, TextLoader
from langchain_core.documents import Document
from pathlib import Path

# 加载
loader = DirectoryLoader(
    "./documents",
    glob="*.txt",
    loader_cls=TextLoader,
    loader_kwargs={'encoding': 'utf-8'}
)
docs = loader.load()

# 增强元数据
enhanced_docs = []
for doc in docs:
    # 添加额外的元数据
    file_path = Path(doc.metadata['source'])

    doc.metadata.update({
        'filename': file_path.name,
        'file_size': file_path.stat().st_size,
        'created_date': file_path.stat().st_mtime,
        'content_length': len(doc.page_content),
        'word_count': len(doc.page_content.split()),
        'category': 'user_docs'  # 自定义分类
    })

    enhanced_docs.append(doc)

# 验证
print(f"元数据示例: {enhanced_docs[0].metadata}")
```

### 支持的文件格式

```python
from langchain_community.document_loaders import (
    TextLoader,              # .txt
    PyPDFLoader,            # .pdf
    UnstructuredMarkdownLoader,  # .md
    CSVLoader,              # .csv
    WebBaseLoader,          # 网页
)

# 对应用法
text_loader = TextLoader("file.txt")
pdf_loader = PyPDFLoader("file.pdf")
csv_loader = CSVLoader("file.csv")
md_loader = UnstructuredMarkdownLoader("file.md")
```

## 2️⃣ RecursiveCharacterTextSplitter - 文本切割

### 作用
递归地按不同分隔符切割文本，保留上下文(overlap)。

### 基础参数详解

```python
from langchain.text_splitter import RecursiveCharacterTextSplitter

splitter = RecursiveCharacterTextSplitter(
    # ===== 关键参数 =====
    chunk_size=500,           # 块大小(字符数)
    chunk_overlap=50,         # 块重叠(保留上下文)

    # ===== 分隔符策略 =====
    separators=[
        "\n\n",   # 优先在段落分隔
        "\n",     # 其次在行分隔
        "。",     # 中文句号
        "!",      # 感叹号
        "?",      # 问号
        "，",     # 中文逗号
        " ",      # 空格
        ""        # 最后按字符分割
    ],

    # ===== 其他参数 =====
    length_function=len,      # 如何计算文本长度
    is_separator_regex=False, # 分隔符是否正则表达式
)
```

### chunk_size的实际影响

```python
from langchain.text_splitter import RecursiveCharacterTextSplitter

# 相同的文本，不同chunk_size
text = "这是很长的文本" * 100

for chunk_size in [200, 500, 1000]:
    splitter = RecursiveCharacterTextSplitter(
        chunk_size=chunk_size,
        chunk_overlap=20
    )

    chunks = splitter.split_text(text)

    print(f"\nchunk_size={chunk_size}:")
    print(f"  块数: {len(chunks)}")
    print(f"  平均块大小: {sum(len(c) for c in chunks) / len(chunks):.0f}")
    print(f"  最小: {min(len(c) for c in chunks)}, 最大: {max(len(c) for c in chunks)}")

# 观察：
# - chunk_size越小，块数越多，精准性高但可能丢失上下文
# - chunk_size越大，块数越少，上下文完整但可能不精准
```

### overlap的重要性

```python
# 演示overlap的作用

text = "...这个概念很重要。 [分界线] 下一个概念..."

# ❌ 无overlap：关键信息被分割
splitter_no_overlap = RecursiveCharacterTextSplitter(
    chunk_size=30,
    chunk_overlap=0  # 无重叠
)

chunks_no = splitter_no_overlap.split_text(text)
# 结果：
# chunk1: "...这个概念很重要。"
# chunk2: "[分界线] 下一个概念..."
# 问题：缺少过渡信息

# ✅ 有overlap：保留关键上下文
splitter_with_overlap = RecursiveCharacterTextSplitter(
    chunk_size=30,
    chunk_overlap=10  # 有重叠
)

chunks_yes = splitter_with_overlap.split_text(text)
# 结果：
# chunk1: "...这个概念很重要。 [分界"
# chunk2: "界线] 下一个概念..."
# 优势：两个chunk都有共同的过渡信息
```

### split_text() vs split_documents()

```python
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_core.documents import Document

splitter = RecursiveCharacterTextSplitter(
    chunk_size=500,
    chunk_overlap=50
)

# ❌ split_text()：只返回字符串，丢失元数据
chunks_text = splitter.split_text("content here")
print(type(chunks_text[0]))  # <class 'str'>

# ✅ split_documents()：返回Document，保留元数据
docs = [Document(
    page_content="content here",
    metadata={'source': 'file.txt'}
)]
chunks_docs = splitter.split_documents(docs)
print(type(chunks_docs[0]))  # <class 'Document'>
print(chunks_docs[0].metadata)  # {'source': 'file.txt'}
```

## 3️⃣ Chroma.from_documents() - 创建向量库

### 作用
一步到位地加载文档、生成embeddings、创建向量库。

### 基础用法

```python
from langchain_community.vectorstores import Chroma
from langchain_openai import OpenAIEmbeddings

# 假设已有chunks
chunks = [
    Document(page_content="文本1", metadata={"source": "file1.txt"}),
    Document(page_content="文本2", metadata={"source": "file2.txt"}),
]

# 一步创建向量库
vectorstore = Chroma.from_documents(
    documents=chunks,
    embedding=OpenAIEmbeddings(model="text-embedding-3-small"),
    persist_directory="./chroma_db"  # 持久化到磁盘
)

print("✅ 向量库创建成功")
```

### 加载已有的向量库

```python
from langchain_community.vectorstores import Chroma
from langchain_openai import OpenAIEmbeddings

# 加载已保存的向量库
vectorstore = Chroma(
    persist_directory="./chroma_db",
    embedding_function=OpenAIEmbeddings(model="text-embedding-3-small")
)

print(f"✅ 向量库加载成功，包含{vectorstore._collection.count()}条记录")
```

### 添加新文档到已有向量库

```python
from langchain_community.vectorstores import Chroma
from langchain_core.documents import Document

# 加载已有的向量库
vectorstore = Chroma(
    persist_directory="./chroma_db",
    embedding_function=OpenAIEmbeddings()
)

# 添加新文档
new_docs = [
    Document(page_content="新内容1", metadata={"source": "new1.txt"}),
    Document(page_content="新内容2", metadata={"source": "new2.txt"}),
]

vectorstore.add_documents(new_docs)

print("✅ 新文档已添加")
```

### 不同Embedding模型的影响

```python
from langchain_community.vectorstores import Chroma
from langchain_openai import OpenAIEmbeddings
from langchain_community.embeddings import HuggingFaceEmbeddings

documents = [...]  # 假设已有文档

# 选项1：OpenAI Embedding(质量最好，有成本)
vectorstore1 = Chroma.from_documents(
    documents,
    OpenAIEmbeddings(model="text-embedding-3-small")
)

# 选项2：本地免费Embedding
vectorstore2 = Chroma.from_documents(
    documents,
    HuggingFaceEmbeddings(
        model_name="sentence-transformers/all-MiniLM-L6-v2"
    )
)

# 选项3：不同的OpenAI Embedding模型
vectorstore3 = Chroma.from_documents(
    documents,
    OpenAIEmbeddings(model="text-embedding-3-large")  # 更强但更贵
)

# 对比：
# - text-embedding-3-small: $0.02/1M tokens (推荐)
# - text-embedding-3-large: $0.13/1M tokens
# - 本地模型: 免费(但质量可能低一些)
```

## 4️⃣ similarity_search() - 相似度检索

### 作用
找到与查询最相关的k个文档。

### 基础用法

```python
from langchain_community.vectorstores import Chroma

vectorstore = Chroma(
    persist_directory="./chroma_db",
    embedding_function=OpenAIEmbeddings()
)

# 最简单的用法
results = vectorstore.similarity_search("什么是AI?", k=3)

print(f"找到{len(results)}个结果:")
for i, doc in enumerate(results):
    print(f"\n结果{i+1}:")
    print(f"  内容: {doc.page_content[:100]}...")
    print(f"  来源: {doc.metadata.get('source')}")
```

### similarity_search_with_scores() - 查看相关度

```python
# 带分数的检索(了解相关度)
results = vectorstore.similarity_search_with_score("查询", k=5)

for doc, score in results:
    print(f"相关度: {score:.4f} (分数越低越相关)")
    print(f"内容: {doc.page_content[:50]}...")

# 使用分数过滤
threshold = 0.5
relevant_results = [
    (doc, score) for doc, score in results
    if score < threshold
]

print(f"高相关度结果: {len(relevant_results)}")
```

### 基于元数据的过滤

```python
# 在检索前过滤
results = vectorstore.similarity_search(
    "查询",
    k=3,
    filter={"source": "important.txt"}  # 只在这个文件中检索
)

# 多条件过滤
results = vectorstore.similarity_search(
    "查询",
    k=3,
    filter={
        "source": "doc1.txt",
        "category": "technical"
    }
)
```

## 5️⃣ 不同检索策略对比

### Similarity Search - 基础相似度

```python
# 最基础、最快的方式
retriever = vectorstore.as_retriever(
    search_type="similarity",
    search_kwargs={"k": 3}
)

results = retriever.get_relevant_documents("查询")
# 返回最相似的3个文档
```

### MMR (Maximum Marginal Relevance) - 多样性优先

```python
# 不仅要相似，还要多样
retriever = vectorstore.as_retriever(
    search_type="mmr",
    search_kwargs={
        "k": 3,           # 最终返回3个
        "fetch_k": 10     # 先取10个，再用MMR筛选
    }
)

# 应用场景：
# - 生成多个不同视角的答案
# - 需要多样化的信息
# - 避免返回重复/相似的结果
```

### Similarity Score Threshold - 质量优先

```python
# 只返回高相关度的结果
retriever = vectorstore.as_retriever(
    search_type="similarity_score_threshold",
    search_kwargs={
        "score_threshold": 0.7,  # 相关度>0.7才返回
        "k": 5                   # 最多返回5个
    }
)

# 应用场景：
# - 严格控制结果质量
# - 宁可少返回也不要低质量
# - 避免"幻觉"(查询无关但硬是返回)
```

### 三种策略的对比

```python
from langchain_community.vectorstores import Chroma

vectorstore = Chroma(...)

query = "什么是机器学习?"

# 策略1：Similarity
results1 = vectorstore.similarity_search(query, k=3)
print(f"Similarity: {len(results1)}个结果，都很相似")

# 策略2：MMR
results2 = vectorstore.as_retriever(
    search_type="mmr",
    search_kwargs={"k": 3, "fetch_k": 10}
).get_relevant_documents(query)
print(f"MMR: {len(results2)}个结果，相似但多样")

# 策略3：Threshold
results3 = vectorstore.as_retriever(
    search_type="similarity_score_threshold",
    search_kwargs={"score_threshold": 0.7, "k": 3}
).get_relevant_documents(query)
print(f"Threshold: {len(results3)}个高质量结果")
```

## 6️⃣ RetrievalQA.from_chain_type() - 完整RAG链

### 作用
一步到位创建完整的检索增强生成链。

### 最基础的RAG

```python
from langchain.chains import RetrievalQA
from langchain_openai import ChatOpenAI
from langchain_community.vectorstores import Chroma

# 创建检索器
retriever = vectorstore.as_retriever(search_kwargs={"k": 3})

# 创建LLM
llm = ChatOpenAI(model="gpt-3.5-turbo", temperature=0)

# 一步创建RAG链
qa_chain = RetrievalQA.from_chain_type(
    llm=llm,
    chain_type="stuff",     # 最简单的类型
    retriever=retriever,
    return_source_documents=True
)

# 使用
result = qa_chain.invoke({"query": "什么是LangChain?"})

print("答案:", result["result"])
print("来源数:", len(result["source_documents"]))
for doc in result["source_documents"]:
    print(f"  - {doc.metadata.get('source')}")
```

### 自定义RAG Prompt

```python
from langchain.chains import RetrievalQA
from langchain.prompts import PromptTemplate
from langchain_openai import ChatOpenAI

# 自定义Prompt
custom_prompt = PromptTemplate(
    template="""基于以下上下文回答问题。

要求:
1. 只使用提供的上下文信息
2. 如果无法从上下文找到答案，明确说"根据提供资料无法回答"
3. 简洁明了，不超过100字
4. 如果引用具体内容，请标注来源

上下文:
{context}

问题: {question}

答案:""",
    input_variables=["context", "question"]
)

qa_chain = RetrievalQA.from_chain_type(
    llm=llm,
    chain_type="stuff",
    retriever=retriever,
    return_source_documents=True,
    chain_type_kwargs={
        "prompt": custom_prompt,
        "document_variable_name": "context"
    }
)
```

### 不同chain_type对比

```python
# chain_type="stuff" - 最简单
# 直接把所有context放入prompt
# 适合: 文档不太长，context数量不多

# chain_type="map_reduce" - 分层处理
# 先对每个document处理，再合并
# 适合: 长文档或大量context

# chain_type="refine" - 迭代改进
# 逐个处理document，不断完善答案
# 适合: 需要精细答案

qa_stuff = RetrievalQA.from_chain_type(
    llm=llm, chain_type="stuff", retriever=retriever
)

qa_map_reduce = RetrievalQA.from_chain_type(
    llm=llm, chain_type="map_reduce", retriever=retriever
)
```

## 📝 速查表

### 何时使用哪个方法？

| 需求 | 推荐方法 | 优势 |
|------|---------|------|
| 单个文件 | `TextLoader` | 简单直接 |
| 多个文件 | `DirectoryLoader` | 批量高效 |
| 切割文本 | `RecursiveCharacterTextSplitter` | 保留上下文 |
| 建立向量库 | `Chroma.from_documents()` | 一步完成 |
| 基础检索 | `similarity_search()` | 快速精准 |
| 多样化检索 | `MMR` | 避免重复 |
| 质量检索 | `score_threshold` | 严格筛选 |
| 完整RAG | `RetrievalQA` | 一体化方案 |

### chunk_size选择速查表

| 场景 | 推荐大小 | 原因 |
|------|---------|------|
| 短问答 | 200-400 | 精准匹配，降低噪声 |
| 一般理解 | 400-800 | 平衡精度和上下文 |
| 长文本 | 800-1500 | 保留充足上下文 |
| 代码文档 | 1000-2000 | 保持代码完整 |
| 技术文档 | 500-1000 | 平衡章节完整 |
