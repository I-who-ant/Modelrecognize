## 1️⃣ get_openai_callback() - 便捷Token追踪

### 作用
使用上下文管理器便捷地统计一段代码中的所有Token使用和成本。

### 最基础的用法

```python
from langchain.callbacks import get_openai_callback
from langchain_openai import ChatOpenAI

llm = ChatOpenAI(model="gpt-3.5-turbo")

# 使用with语句追踪
with get_openai_callback() as cb:
    # 单个调用
    response = llm.invoke("什么是LangChain?")

    # 统计信息会自动累计
    print(f"Prompt Tokens: {cb.prompt_tokens}")
    print(f"Completion Tokens: {cb.completion_tokens}")
    print(f"Total Tokens: {cb.total_tokens}")
    print(f"Total Cost: ${cb.total_cost:.6f}")
    print(f"Successful Requests: {cb.successful_requests}")
```

### 追踪多个调用

```python
from langchain.callbacks import get_openai_callback
from langchain_openai import ChatOpenAI

llm = ChatOpenAI(model="gpt-3.5-turbo")

with get_openai_callback() as cb:
    # 调用1
    llm.invoke("什么是LangChain?")

    # 调用2
    llm.invoke("LangChain有什么优势?")

    # 调用3
    llm.invoke("如何开始使用LangChain?")

    # 所有调用的Token都被累计
    print(f"共调用了{cb.successful_requests}次")
    print(f"总Token数: {cb.total_tokens}")
    print(f"总成本: ${cb.total_cost:.6f}")

    # 也可以查看各部分
    avg_tokens = cb.total_tokens / cb.successful_requests
    print(f"平均每次调用: {avg_tokens:.0f} tokens")
```

### 追踪流式调用

```python
from langchain.callbacks import get_openai_callback
from langchain_openai import ChatOpenAI

llm = ChatOpenAI(model="gpt-3.5-turbo")

with get_openai_callback() as cb:
    # 流式输出不影响Token计数
    for chunk in llm.stream("写一首诗"):
        print(chunk.content, end="", flush=True)

    print(f"\n\n流式调用的Token数: {cb.total_tokens}")
    # 流式总Token数 = invoke()的Token数
    # 只是返回方式不同而已
```

### 成本统计示例

```python
from langchain.callbacks import get_openai_callback
from langchain_openai import ChatOpenAI

# 定义不同模型的价格(每百万token的美元)
PRICES = {
    "gpt-3.5-turbo": {"input": 0.5, "output": 1.5},
    "gpt-4-turbo": {"input": 10.0, "output": 30.0},
    "gpt-4": {"input": 30.0, "output": 60.0},
}

def estimate_cost(model: str, prompt_tokens: int, completion_tokens: int) -> float:
    """估算API成本"""
    price = PRICES.get(model, PRICES["gpt-3.5-turbo"])
    cost = (
        prompt_tokens * price["input"] +
        completion_tokens * price["output"]
    ) / 1_000_000
    return cost

# 追踪使用
llm_35 = ChatOpenAI(model="gpt-3.5-turbo")
llm_4 = ChatOpenAI(model="gpt-4-turbo")

questions = [
    "什么是AI?",
    "什么是ML?",
    "什么是DL?"
]

print("=== GPT-3.5-turbo ===")
with get_openai_callback() as cb:
    for q in questions:
        llm_35.invoke(q)

    cost_35 = estimate_cost(
        "gpt-3.5-turbo",
        cb.prompt_tokens,
        cb.completion_tokens
    )
    print(f"Token: {cb.total_tokens}")
    print(f"预估成本: ${cost_35:.6f}")

print("\n=== GPT-4-turbo ===")
with get_openai_callback() as cb:
    for q in questions:
        llm_4.invoke(q)

    cost_4 = estimate_cost(
        "gpt-4-turbo",
        cb.prompt_tokens,
        cb.completion_tokens
    )
    print(f"Token: {cb.total_tokens}")
    print(f"预估成本: ${cost_4:.6f}")

print(f"\n成本差异: {cost_4/cost_35:.1f}倍")
```

## 2️⃣ stream() 和 astream() - 流式输出

### 作用
实时返回LLM的生成过程，不需要等待完整响应。

### stream() - 同步流式

```python
from langchain_openai import ChatOpenAI

llm = ChatOpenAI(model="gpt-3.5-turbo")

# 基础用法
print("生成中: ", end="", flush=True)
for chunk in llm.stream("写一首关于春天的诗"):
    print(chunk.content, end="", flush=True)
print()  # 换行
```

### 收集stream()的chunks

```python
from langchain_openai import ChatOpenAI

llm = ChatOpenAI(model="gpt-3.5-turbo")

# 方式1: 实时输出+收集
full_text = ""
chunk_count = 0

for chunk in llm.stream("解释什么是装饰器"):
    full_text += chunk.content
    chunk_count += 1
    print(chunk.content, end="", flush=True)

print(f"\n\n总共{chunk_count}个chunk")

# 方式2: 先列表后处理
chunks = list(llm.stream("解释什么是装饰器"))
print(f"总共{len(chunks)}个chunk")

for i, chunk in enumerate(chunks):
    print(f"Chunk {i+1}: {chunk.content}")
```

### astream() - 异步流式

```python
import asyncio
from langchain_openai import ChatOpenAI

llm = ChatOpenAI(model="gpt-3.5-turbo")

async def main():
    print("异步流式生成中: ", end="", flush=True)

    async for chunk in llm.astream("写一首诗"):
        print(chunk.content, end="", flush=True)

    print()

# 运行
asyncio.run(main())
```

### 流式 vs 非流式的核心区别

```python
import time
from langchain_openai import ChatOpenAI

llm = ChatOpenAI(model="gpt-3.5-turbo")
question = "详细介绍人工智能的发展历史" * 2

# ❌ 非流式: invoke()
print("=== 非流式 (invoke) ===")
start = time.time()

print("正在等待...")
response = llm.invoke(question)
elapsed = time.time() - start

print(f"等待了{elapsed:.1f}秒才看到回复:")
print(response.content)

# ✅ 流式: stream()
print("\n=== 流式 (stream) ===")
start = time.time()

print("生成中: ", end="", flush=True)
for chunk in llm.stream(question):
    print(chunk.content, end="", flush=True)
elapsed = time.time() - start

print(f"\n耗时{elapsed:.1f}秒")

# 重要：Token数和成本完全相同，但体验不同！
```

### StreamingStdOutCallbackHandler - 自动流式输出

```python
from langchain_openai import ChatOpenAI
from langchain.callbacks.streaming_stdout import StreamingStdOutCallbackHandler

# 在初始化时使用流式回调
llm = ChatOpenAI(
    model="gpt-3.5-turbo",
    streaming=True,
    callbacks=[StreamingStdOutCallbackHandler()]
)

# 这样invoke()就会自动流式输出
response = llm.invoke("写一首诗")

# 同时也能获取完整响应
print(f"\n完整文本长度: {len(response.content)}")
```

## 3️⃣ 自定义BaseCallbackHandler - 高级Token追踪

### 作用
深度定制Token追踪逻辑，比get_openai_callback()更灵活。

### 基础实现

```python
from langchain.callbacks.base import BaseCallbackHandler
from langchain_openai import ChatOpenAI
from typing import Any, Dict

class TokenCounter(BaseCallbackHandler):
    """自定义Token计数器"""

    def __init__(self):
        self.total_tokens = 0
        self.prompt_tokens = 0
        self.completion_tokens = 0
        self.request_count = 0

    def on_llm_end(self, response: Any, **kwargs: Any) -> None:
        """LLM调用结束时被调用"""
        if hasattr(response, 'llm_output') and response.llm_output:
            token_usage = response.llm_output.get('token_usage', {})

            self.prompt_tokens += token_usage.get('prompt_tokens', 0)
            self.completion_tokens += token_usage.get('completion_tokens', 0)
            self.total_tokens += token_usage.get('total_tokens', 0)
            self.request_count += 1

    def report(self):
        """输出报告"""
        print(f"\n=== Token使用报告 ===")
        print(f"总请求数: {self.request_count}")
        print(f"总Token数: {self.total_tokens}")
        print(f"  Prompt: {self.prompt_tokens}")
        print(f"  Completion: {self.completion_tokens}")

# 使用
counter = TokenCounter()
llm = ChatOpenAI(model="gpt-3.5-turbo", callbacks=[counter])

llm.invoke("什么是AI?")
llm.invoke("什么是ML?")
llm.invoke("什么是DL?")

counter.report()
```

### 高级：追踪每个请求的详情

```python
from langchain.callbacks.base import BaseCallbackHandler
from langchain_openai import ChatOpenAI
from typing import Any, Dict, List

class DetailedTokenCounter(BaseCallbackHandler):
    """追踪每个请求的详细Token信息"""

    def __init__(self):
        self.request_details = []
        self.current_request = None

    def on_llm_start(self, serialized: Dict[str, Any], prompts: List[str], **kwargs) -> None:
        """LLM开始调用时"""
        self.current_request = {
            "prompt": prompts[0] if prompts else "",
            "prompt_length": len(prompts[0]) if prompts else 0,
        }

    def on_llm_end(self, response: Any, **kwargs: Any) -> None:
        """LLM调用结束时"""
        if hasattr(response, 'llm_output') and response.llm_output:
            token_usage = response.llm_output.get('token_usage', {})

            self.current_request.update({
                "prompt_tokens": token_usage.get('prompt_tokens', 0),
                "completion_tokens": token_usage.get('completion_tokens', 0),
                "total_tokens": token_usage.get('total_tokens', 0),
            })

            self.request_details.append(self.current_request)

    def report(self):
        """详细报告"""
        print(f"\n=== 详细Token报告 ===")
        for i, detail in enumerate(self.request_details, 1):
            print(f"\n请求 {i}:")
            print(f"  Prompt: {detail.get('prompt', '')[:50]}...")
            print(f"  Token数: {detail.get('total_tokens', 0)}")
            print(f"    - Prompt: {detail.get('prompt_tokens', 0)}")
            print(f"    - Completion: {detail.get('completion_tokens', 0)}")

# 使用
counter = DetailedTokenCounter()
llm = ChatOpenAI(model="gpt-3.5-turbo", callbacks=[counter])

for q in ["什么是AI?", "什么是ML?", "什么是DL?"]:
    llm.invoke(q)

counter.report()
```

## 4️⃣ PydanticOutputParser - 结构化输出

### 作用
将LLM的输出解析为Python对象，实现类型安全的结构化数据。

### 基础用法

```python
from pydantic import BaseModel, Field
from langchain_core.output_parsers import PydanticOutputParser
from langchain_openai import ChatOpenAI

# 定义数据模型
class Person(BaseModel):
    name: str = Field(description="人名")
    age: int = Field(description="年龄")
    city: str = Field(description="城市")

# 创建解析器
parser = PydanticOutputParser(pydantic_object=Person)

# 获取格式说明
format_instructions = parser.get_format_instructions()
print("格式说明:")
print(format_instructions)

# 创建Prompt
llm = ChatOpenAI(model="gpt-3.5-turbo", temperature=0)

prompt = f"""从以下文本中提取人物信息:

张三是一个25岁的软件工程师，居住在北京。

{format_instructions}"""

# 调用LLM
response = llm.invoke(prompt)

# 解析为Python对象
person = parser.parse(response.content)

print(f"姓名: {person.name}")
print(f"年龄: {person.age}")
print(f"城市: {person.city}")
print(f"类型: {type(person)}")  # <class '__main__.Person'>
```

### 复杂模型（列表、验证）

```python
from pydantic import BaseModel, Field, validator
from langchain_core.output_parsers import PydanticOutputParser
from langchain_openai import ChatOpenAI
from typing import List

# 复杂模型
class ProductReview(BaseModel):
    product_name: str = Field(description="产品名称")
    rating: int = Field(description="评分1-5", ge=1, le=5)  # 验证范围
    pros: List[str] = Field(description="优点列表")
    cons: List[str] = Field(description="缺点列表")
    recommended: bool = Field(description="是否推荐")

    @validator('pros', 'cons')
    def validate_list_not_empty(cls, v):
        if not v:
            raise ValueError('列表不能为空')
        return v

# 使用
parser = PydanticOutputParser(pydantic_object=ProductReview)
llm = ChatOpenAI(model="gpt-3.5-turbo", temperature=0)

prompt = f"""分析以下产品评价:

iPhone 15 Pro
优点: 屏幕清晰, 性能强劲, 续航给力
缺点: 价格昂贵, 过热
评分: 4/5

{parser.get_format_instructions()}"""

response = llm.invoke(prompt)
review = parser.parse(response.content)

print(f"产品: {review.product_name}")
print(f"评分: {review.rating}/5")
print(f"优点: {', '.join(review.pros)}")
print(f"缺点: {', '.join(review.cons)}")
print(f"推荐: {'是' if review.recommended else '否'}")
```

## 5️⃣ OutputFixingParser - 错误自动修复

### 作用
当LLM输出格式不对时，自动调用LLM修复，提高成功率。

### 基础用法

```python
from pydantic import BaseModel, Field
from langchain_core.output_parsers import PydanticOutputParser, OutputFixingParser
from langchain_openai import ChatOpenAI

class Person(BaseModel):
    name: str = Field(description="姓名")
    age: int = Field(description="年龄")

# 创建原始解析器
pydantic_parser = PydanticOutputParser(pydantic_object=Person)

# 包装为修复解析器
fixing_parser = OutputFixingParser.from_llm(
    parser=pydantic_parser,
    llm=ChatOpenAI(model="gpt-3.5-turbo", temperature=0)
)

# 故意给个错的输入
bad_json = """
{
    "name": "王五",
    "age": "三十岁"
}"""

print("错误的JSON:")
print(bad_json)

try:
    # 普通解析器会失败
    print("\n尝试普通解析...")
    pydantic_parser.parse(bad_json)
except Exception as e:
    print(f"❌ 失败: {str(e)[:50]}...")

try:
    # OutputFixingParser会修复
    print("\n尝试修复解析...")
    person = fixing_parser.parse(bad_json)
    print(f"✅ 修复成功!")
    print(f"  姓名: {person.name}")
    print(f"  年龄: {person.age}")
except Exception as e:
    print(f"❌ 修复也失败: {e}")
```

### 原理：修复过程

```
错误的JSON输入
    ↓
OutputFixingParser.parse()
    ↓
尝试用原始Parser解析
    ↓
如果失败，提示错误给LLM
    ↓
LLM修复JSON格式
    ↓
再次尝试解析修复后的结果
    ↓
返回成功的Python对象
```

## 6️⃣ OutputParser对比

### 作用
不同场景使用不同的输出解析器。

### 三种主流Parser对比

```python
from langchain_core.output_parsers import (
    PydanticOutputParser,
    JsonOutputParser,
    StrOutputParser
)
from langchain.output_parsers import StructuredOutputParser, ResponseSchema
from pydantic import BaseModel

# Parser 1: PydanticOutputParser
class PersonPydantic(BaseModel):
    name: str
    age: int

parser1 = PydanticOutputParser(pydantic_object=PersonPydantic)

# Parser 2: JsonOutputParser (返回dict)
parser2 = JsonOutputParser()

# Parser 3: StructuredOutputParser (固定字段)
response_schemas = [
    ResponseSchema(name="name", description="姓名"),
    ResponseSchema(name="age", description="年龄"),
]
parser3 = StructuredOutputParser.from_response_schemas(response_schemas)

# Parser 4: StrOutputParser (返回字符串)
parser4 = StrOutputParser()

# 对比
print("=== OutputParser对比 ===\n")

outputs = {
    "Pydantic": {
        "优点": ["类型安全", "自动验证", "IDE补全"],
        "缺点": ["需要定义类"],
        "场景": "复杂数据结构"
    },
    "Json": {
        "优点": ["灵活", "简单"],
        "缺点": ["无类型检查"],
        "场景": "简单数据"
    },
    "Structured": {
        "优点": ["配置简单"],
        "缺点": ["功能有限"],
        "场景": "固定字段"
    },
    "Str": {
        "优点": ["最灵活"],
        "缺点": ["需要手动处理"],
        "场景": "自由文本"
    }
}

for parser_type, info in outputs.items():
    print(f"{parser_type}:")
    print(f"  优点: {', '.join(info['优点'])}")
    print(f"  缺点: {', '.join(info['缺点'])}")
    print(f"  场景: {info['场景']}\n")
```

### 选择建议

```python
# 场景1: 严格的数据模型 → PydanticOutputParser
class Invoice(BaseModel):
    invoice_no: str
    amount: float
    date: str
    items: List[str]

parser = PydanticOutputParser(pydantic_object=Invoice)

# 场景2: 灵活的数据 → JsonOutputParser
parser = JsonOutputParser()

# 场景3: 简单字段 → StructuredOutputParser
schemas = [
    ResponseSchema(name="summary", description="总结"),
    ResponseSchema(name="keywords", description="关键词"),
]
parser = StructuredOutputParser.from_response_schemas(schemas)

# 场景4: 纯文本 → StrOutputParser
parser = StrOutputParser()
```

## 📝 速查表

### 何时使用哪个工具？

| 需求 | 推荐工具 | 原因 |
|------|---------|------|
| 统计Token成本 | `get_openai_callback()` | 最简洁 |
| 实时输出反馈 | `stream()` | 用户体验好 |
| 深度定制追踪 | `自定义Callback` | 灵活性强 |
| 结构化数据 | `PydanticOutputParser` | 类型安全 |
| 修复格式错误 | `OutputFixingParser` | 提高成功率 |

### 工具组合最佳实践

```python
from langchain.callbacks import get_openai_callback
from langchain_core.output_parsers import PydanticOutputParser, OutputFixingParser
from langchain_openai import ChatOpenAI
from pydantic import BaseModel

class Result(BaseModel):
    answer: str
    confidence: float

llm = ChatOpenAI(model="gpt-3.5-turbo", temperature=0)

# 组合：追踪 + 流式 + 结构化 + 修复
pydantic_parser = PydanticOutputParser(pydantic_object=Result)
fixing_parser = OutputFixingParser.from_llm(parser=pydantic_parser, llm=llm)

with get_openai_callback() as cb:
    # 使用固定流式+解析
    for chunk in llm.stream("问题"):
        print(chunk.content, end="", flush=True)

    response = llm.invoke("问题")
    result = fixing_parser.parse(response.content)

    print(f"\n\n答案: {result.answer}")
    print(f"信心度: {result.confidence}")
    print(f"成本: ${cb.total_cost:.6f}")
```
